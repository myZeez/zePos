class ZePosApp extends StatefulWidget {
  const ZePosApp({super.key, this.database});

  final ZePosDatabase? database;

  @override
  State<ZePosApp> createState() => _ZePosAppState();
}

class _ZePosAppState extends State<ZePosApp> {
  late Future<ZePosDatabase> _databaseFuture;
  var _isAuthenticated = false;

  @override
  void initState() {
    super.initState();
    _openDatabase();
  }

  void _openDatabase() {
    _databaseFuture = widget.database == null
        ? ZePosDatabase.open()
        : Future<ZePosDatabase>.value(widget.database);
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'ZePOS',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        useMaterial3: true,
        colorScheme: ColorScheme.fromSeed(
          seedColor: ZeColors.primary,
          brightness: Brightness.light,
          surface: ZeColors.surface,
        ),
        scaffoldBackgroundColor: ZeColors.background,
        textTheme: ThemeData.light().textTheme
            .apply(
              bodyColor: ZeColors.body,
              displayColor: ZeColors.ink,
              fontFamily: 'Arial',
            )
            .copyWith(
              headlineSmall: ZeText.pageTitle,
              titleLarge: ZeText.sectionTitle,
              titleMedium: ZeText.cardTitle,
              bodyLarge: const TextStyle(fontSize: 16, height: 1.5),
              bodyMedium: ZeText.body,
              bodySmall: ZeText.meta,
              labelLarge: const TextStyle(
                fontSize: 14,
                fontWeight: FontWeight.w600,
              ),
              labelMedium: ZeText.label,
            ),
        appBarTheme: const AppBarTheme(
          backgroundColor: ZeColors.background,
          foregroundColor: ZeColors.ink,
          elevation: 0,
          scrolledUnderElevation: 0,
          surfaceTintColor: Colors.transparent,
        ),
        filledButtonTheme: FilledButtonThemeData(
          style: FilledButton.styleFrom(
            minimumSize: const Size(44, 46),
            backgroundColor: ZeColors.primary,
            foregroundColor: Colors.white,
            shape: RoundedRectangleBorder(borderRadius: ZeRadius.control),
            textStyle: const TextStyle(
              fontWeight: FontWeight.w700,
              letterSpacing: 0.15,
            ),
            elevation: 0,
          ),
        ),
        outlinedButtonTheme: OutlinedButtonThemeData(
          style: OutlinedButton.styleFrom(
            minimumSize: const Size(44, 46),
            foregroundColor: ZeColors.primaryDark,
            side: const BorderSide(color: ZeColors.primary, width: 1.25),
            shape: RoundedRectangleBorder(borderRadius: ZeRadius.control),
          ),
        ),
        inputDecorationTheme: InputDecorationTheme(
          filled: true,
          fillColor: ZeColors.surface,
          contentPadding: const EdgeInsets.symmetric(
            horizontal: 14,
            vertical: 14,
          ),
          border: const OutlineInputBorder(
            borderRadius: ZeRadius.control,
            borderSide: BorderSide(color: ZeColors.border),
          ),
          enabledBorder: const OutlineInputBorder(
            borderRadius: ZeRadius.control,
            borderSide: BorderSide(color: ZeColors.border),
          ),
          focusedBorder: const OutlineInputBorder(
            borderRadius: ZeRadius.control,
            borderSide: BorderSide(color: ZeColors.primary, width: 2),
          ),
        ),
        dividerTheme: const DividerThemeData(
          color: ZeColors.borderSoft,
          thickness: 1,
          space: 1,
        ),
        cardTheme: const CardThemeData(
          color: ZeColors.surface,
          elevation: 0,
          margin: EdgeInsets.zero,
          shape: RoundedRectangleBorder(
            borderRadius: ZeRadius.panel,
            side: BorderSide(color: ZeColors.border),
          ),
        ),
        navigationBarTheme: NavigationBarThemeData(
          backgroundColor: ZeColors.surface,
          indicatorColor: ZeColors.surfaceSelected,
          labelTextStyle: WidgetStateProperty.resolveWith(
            (states) => TextStyle(
              color: states.contains(WidgetState.selected)
                  ? ZeColors.primaryDark
                  : ZeColors.navigationInk,
              fontSize: 12,
              fontWeight: FontWeight.w600,
            ),
          ),
        ),
      ),
      home: FutureBuilder<ZePosDatabase>(
        future: _databaseFuture,
        builder: (context, snapshot) {
          if (snapshot.connectionState != ConnectionState.done) {
            return const _AppLoadingPage();
          }
          if (snapshot.hasError || !snapshot.hasData) {
            return _AppDatabaseErrorPage(
              onRetry: () => setState(_openDatabase),
            );
          }
          return FutureBuilder<StoreProfile?>(
            future: snapshot.data!.loadStoreProfile(),
            builder: (context, profileSnapshot) {
              if (profileSnapshot.connectionState != ConnectionState.done) {
                return const _AppLoadingPage();
              }
              if (profileSnapshot.hasError) {
                return _AppDatabaseErrorPage(
                  onRetry: () => setState(_openDatabase),
                );
              }
              if (profileSnapshot.data == null) {
                return OnboardingPage(
                  database: snapshot.data!,
                  onCompleted: () => setState(() {
                    _isAuthenticated = false;
                    _openDatabase();
                  }),
                );
              }
              return FutureBuilder<CredentialInfo?>(
                future: snapshot.data!.loadCredentialInfo(),
                builder: (context, credentialSnapshot) {
                  if (credentialSnapshot.connectionState !=
                      ConnectionState.done) {
                    return const _AppLoadingPage();
                  }
                  if (credentialSnapshot.hasError) {
                    return _AppDatabaseErrorPage(
                      onRetry: () => setState(_openDatabase),
                    );
                  }
                  final credentials = credentialSnapshot.data;
                  if (credentials == null) {
                    return CredentialSetupPage(
                      database: snapshot.data!,
                      onCompleted: () => setState(_openDatabase),
                    );
                  }
                  if (!_isAuthenticated) {
                    return LoginPage(
                      database: snapshot.data!,
                      onAuthenticated: () =>
                          setState(() => _isAuthenticated = true),
                    );
                  }
                  if (credentials.mustChangePassword) {
                    return ChangePasswordPage(
                      database: snapshot.data!,
                      username: credentials.username,
                      onCompleted: () => setState(_openDatabase),
                    );
                  }
                  return ZePosShell(
                    database: snapshot.data!,
                    onDatabaseRestored: () => setState(_openDatabase),
                  );
                },
              );
            },
          );
        },
      ),
    );
  }
}

class OnboardingPage extends StatefulWidget {
  const OnboardingPage({
    super.key,
    required this.database,
    required this.onCompleted,
  });

  final ZePosDatabase database;
  final VoidCallback onCompleted;

  @override
  State<OnboardingPage> createState() => _OnboardingPageState();
}

class _OnboardingPageState extends State<OnboardingPage> {
  final _formKey = GlobalKey<FormState>();
  final _storeNameController = TextEditingController();
  final _ownerNameController = TextEditingController();
  final _addressController = TextEditingController();
  final _phoneController = TextEditingController();
  var _isSaving = false;
  String? _error;

  @override
  void dispose() {
    _storeNameController.dispose();
    _ownerNameController.dispose();
    _addressController.dispose();
    _phoneController.dispose();
    super.dispose();
  }

  String? _required(String? value, String label) {
    if (value == null || value.trim().isEmpty) return '$label wajib diisi.';
    return null;
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() {
      _isSaving = true;
      _error = null;
    });
    try {
      await widget.database.saveStoreProfile(
        StoreProfile(
          storeName: _storeNameController.text.trim(),
          ownerName: _ownerNameController.text.trim(),
          address: _addressController.text.trim(),
          phone: _phoneController.text.trim(),
        ),
      );
      if (!mounted) return;
      widget.onCompleted();
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSaving = false;
        _error = 'Data toko belum dapat disimpan. Coba lagi.';
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 480),
              child: Form(
                key: _formKey,
                child: Container(
                  padding: const EdgeInsets.all(28),
                  decoration: BoxDecoration(
                    color: Colors.white,
                    borderRadius: ZeRadius.panel,
                    border: Border.all(color: ZeColors.borderSoft),
                  ),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      const _BrandLockup(compact: false),
                      const SizedBox(height: 32),
                      const Text(
                        'Selamat datang di ZePOS',
                        style: TextStyle(
                          fontSize: 24,
                          fontWeight: FontWeight.w600,
                        ),
                      ),
                      const SizedBox(height: 8),
                      const Text(
                        'Lengkapi data toko sekali saja. Data ini dipakai pada nota, laporan, backup, dan pemulihan akun.',
                        style: TextStyle(color: ZeColors.subtle, height: 1.45),
                      ),
                      const SizedBox(height: 24),
                      _FormFieldGroup(
                        label: 'Nama usaha',
                        child: TextFormField(
                          controller: _storeNameController,
                          textCapitalization: TextCapitalization.words,
                          validator: (value) => _required(value, 'Nama usaha'),
                          decoration: const InputDecoration(
                            hintText: 'Contoh: Toko Maju Jaya',
                          ),
                        ),
                      ),
                      const SizedBox(height: 16),
                      _FormFieldGroup(
                        label: 'Nama pemilik',
                        child: TextFormField(
                          controller: _ownerNameController,
                          textCapitalization: TextCapitalization.words,
                          validator: (value) =>
                              _required(value, 'Nama pemilik'),
                          decoration: const InputDecoration(
                            hintText: 'Contoh: Budi Aulyansyah',
                          ),
                        ),
                      ),
                      const SizedBox(height: 16),
                      _FormFieldGroup(
                        label: 'Nomor telepon toko',
                        child: TextFormField(
                          controller: _phoneController,
                          keyboardType: TextInputType.phone,
                          validator: (value) =>
                              _required(value, 'Nomor telepon toko'),
                          decoration: const InputDecoration(
                            hintText: 'Contoh: 0812 3456 7890',
                            helperText:
                                'Dipakai untuk konfirmasi pemulihan akun.',
                          ),
                        ),
                      ),
                      const SizedBox(height: 16),
                      _FormFieldGroup(
                        label: 'Alamat usaha',
                        child: TextFormField(
                          controller: _addressController,
                          minLines: 2,
                          maxLines: 3,
                          textCapitalization: TextCapitalization.sentences,
                          validator: (value) =>
                              _required(value, 'Alamat usaha'),
                          decoration: const InputDecoration(
                            hintText: 'Alamat lengkap usaha',
                          ),
                        ),
                      ),
                      if (_error case final String message) ...[
                        const SizedBox(height: 12),
                        Text(
                          message,
                          style: const TextStyle(color: ZeColors.danger),
                        ),
                      ],
                      const SizedBox(height: 24),
                      SizedBox(
                        width: double.infinity,
                        child: FilledButton.icon(
                          onPressed: _isSaving ? null : _save,
                          icon: _isSaving
                              ? const SizedBox(
                                  width: 18,
                                  height: 18,
                                  child: CircularProgressIndicator(
                                    strokeWidth: 2,
                                    color: Colors.white,
                                  ),
                                )
                              : const Icon(Icons.arrow_forward),
                          label: Text(
                            _isSaving ? 'Menyimpan...' : 'Mulai gunakan ZePOS',
                          ),
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }
}

class LoginPage extends StatefulWidget {
  const LoginPage({
    super.key,
    required this.database,
    required this.onAuthenticated,
  });

  final ZePosDatabase database;
  final VoidCallback onAuthenticated;

  @override
  State<LoginPage> createState() => _LoginPageState();
}

class _LoginPageState extends State<LoginPage> {
  final _formKey = GlobalKey<FormState>();
  final _usernameController = TextEditingController();
  final _passwordController = TextEditingController();
  var _isSubmitting = false;
  var _showPassword = false;
  String? _error;

  @override
  void dispose() {
    _usernameController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  Future<void> _login() async {
    if (_isSubmitting) return;
    if (!_formKey.currentState!.validate()) return;
    setState(() {
      _isSubmitting = true;
      _error = null;
    });
    final credentials = await widget.database.verifyCredentials(
      username: _usernameController.text,
      password: _passwordController.text,
    );
    if (!mounted) return;
    if (credentials == null) {
      setState(() {
        _isSubmitting = false;
        _error = 'Username atau password tidak sesuai.';
      });
      return;
    }
    widget.onAuthenticated();
  }

  Future<void> _recoverAccount() async {
    await showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) => _RecoverAccountSheet(database: widget.database),
    );
  }

  @override
  Widget build(BuildContext context) {
    return _AuthScaffold(
      title: 'Masuk ke ZePOS',
      subtitle: 'Gunakan akun lokal toko untuk melanjutkan.',
      child: Form(
        key: _formKey,
        child: Column(
          children: [
            _AuthField(
              label: 'Username',
              controller: _usernameController,
              textInputAction: TextInputAction.next,
              validator: (value) => value == null || value.trim().isEmpty
                  ? 'Masukkan username.'
                  : null,
            ),
            const SizedBox(height: 16),
            _AuthField(
              label: 'Password',
              controller: _passwordController,
              obscure: !_showPassword,
              textInputAction: TextInputAction.done,
              onSubmitted: (_) => _login(),
              suffixIcon: IconButton(
                tooltip: _showPassword
                    ? 'Sembunyikan password'
                    : 'Tampilkan password',
                onPressed: _isSubmitting
                    ? null
                    : () => setState(() => _showPassword = !_showPassword),
                icon: Icon(
                  _showPassword
                      ? Icons.visibility_off_outlined
                      : Icons.visibility_outlined,
                ),
              ),
              validator: (value) =>
                  value == null || value.isEmpty ? 'Masukkan password.' : null,
            ),
            if (_error case final String message) ...[
              const SizedBox(height: 12),
              Text(message, style: const TextStyle(color: ZeColors.danger)),
            ],
            const SizedBox(height: 24),
            _AuthPrimaryButton(
              label: _isSubmitting ? 'Memeriksa...' : 'Masuk',
              loading: _isSubmitting,
              onPressed: _isSubmitting ? null : _login,
            ),
            const SizedBox(height: 8),
            TextButton(
              onPressed: _isSubmitting ? null : _recoverAccount,
              child: const Text('Lupa username atau password?'),
            ),
          ],
        ),
      ),
    );
  }
}

class _RecoverAccountSheet extends StatefulWidget {
  const _RecoverAccountSheet({required this.database});

  final ZePosDatabase database;

  @override
  State<_RecoverAccountSheet> createState() => _RecoverAccountSheetState();
}

class _RecoverAccountSheetState extends State<_RecoverAccountSheet> {
  final _phoneController = TextEditingController();
  var _isSubmitting = false;
  String? _message;

  @override
  void dispose() {
    _phoneController.dispose();
    super.dispose();
  }

  Future<void> _recover() async {
    setState(() {
      _isSubmitting = true;
      _message = null;
    });
    final username = await widget.database.recoverAccount(
      _phoneController.text,
    );
    if (!mounted) return;
    setState(() {
      _isSubmitting = false;
      _message = username == null
          ? 'Nomor telepon tidak cocok dengan profil toko.'
          : 'Akun $username dipulihkan. Gunakan password sementara password123, lalu buat password baru.';
    });
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Padding(
        padding: EdgeInsets.fromLTRB(
          24,
          8,
          24,
          24 + MediaQuery.viewInsetsOf(context).bottom,
        ),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text(
              'Pulihkan Akun',
              style: TextStyle(fontSize: 22, fontWeight: FontWeight.w600),

class ZePosShell extends StatefulWidget {
  const ZePosShell({
    super.key,
    required this.database,
    required this.onDatabaseRestored,
  });

  final ZePosDatabase database;
  final VoidCallback onDatabaseRestored;

  @override
  State<ZePosShell> createState() => _ZePosShellState();
}

class _ZePosShellState extends State<ZePosShell> {
  int _activeSection = 1;
  int _dataVersion = 0;
  String _scannerBuffer = '';
  String _scannerQuery = '';
  Timer? _scannerTimer;

  @override
  void dispose() {
    _scannerTimer?.cancel();
    super.dispose();
  }

  void _selectSection(int index) {
    setState(() {
      _activeSection = index;
      _dataVersion++;
    });
  }

  KeyEventResult _listenForBarcode(FocusNode node, KeyEvent event) {
    if (event is! KeyDownEvent) return KeyEventResult.ignored;
    final focusedWidget = FocusManager.instance.primaryFocus?.context?.widget;
    if (focusedWidget is EditableText) return KeyEventResult.ignored;
    if (event.logicalKey == LogicalKeyboardKey.enter) {
      if (_scannerBuffer.length >= 6) {
        setState(() {
          _scannerQuery = _scannerBuffer;
          _scannerBuffer = '';
          _activeSection = 1;
          _dataVersion++;
        });
        _scannerTimer?.cancel();
        return KeyEventResult.handled;
      }
      _scannerBuffer = '';
      return KeyEventResult.ignored;
    }
    final character = event.character;
    if (character == null || !RegExp(r'[0-9A-Za-z\-]').hasMatch(character)) {
      return KeyEventResult.ignored;
    }
    _scannerBuffer += character;
    _scannerTimer?.cancel();
    _scannerTimer = Timer(const Duration(milliseconds: 400), () {
      _scannerBuffer = '';
    });
    return KeyEventResult.ignored;
  }

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        final mobile = constraints.maxWidth < 720;
        return Scaffold(
          body: SafeArea(
            child: Focus(
              autofocus: true,
              onKeyEvent: _listenForBarcode,
              child: Column(
                children: [
                  if (mobile)
                    _MobileTopBar(
                      database: widget.database,
                      dataVersion: _dataVersion,
                      onViewStockAlerts: () => _selectSection(3),
                    )
                  else
                    _DesktopSidebar(
                      activeIndex: _activeSection,
                      onChanged: _selectSection,
                      database: widget.database,
                      dataVersion: _dataVersion,
                      onViewStockAlerts: () => _selectSection(3),
                    ),
                  Expanded(child: _buildContent(_activeSection)),
                ],
              ),
            ),
          ),
          bottomNavigationBar: mobile
              ? NavigationBar(
                  selectedIndex: _activeSection <= 3 ? _activeSection : 4,
                  onDestinationSelected: (index) {
                    if (index < 4) {
                      _selectSection(index);
                    } else {
                      _openMobileMoreMenu();
                    }
                  },
                  destinations: const [
                    NavigationDestination(
                      icon: Icon(Icons.home_outlined),
                      selectedIcon: Icon(Icons.home),
                      label: 'Beranda',
                    ),
                    NavigationDestination(
                      icon: Icon(Icons.point_of_sale_outlined),
                      selectedIcon: Icon(Icons.point_of_sale),
                      label: 'Kasir',
                    ),
                    NavigationDestination(
                      icon: Icon(Icons.inventory_2_outlined),
                      selectedIcon: Icon(Icons.inventory_2),
                      label: 'Produk',
                    ),
                    NavigationDestination(
                      icon: Icon(Icons.warehouse_outlined),
                      selectedIcon: Icon(Icons.warehouse),
                      label: 'Stok',
                    ),
                    NavigationDestination(
                      icon: Icon(Icons.more_horiz),
                      label: 'Lainnya',
                    ),
                  ],
                )
              : null,
        );
      },
    );
  }

  Future<void> _openMobileMoreMenu() async {
    final selected = await showModalBottomSheet<int>(
      context: context,
      showDragHandle: true,
      builder: (context) => SafeArea(
        child: Padding(
          padding: const EdgeInsets.fromLTRB(
            ZeSpace.md,
            0,
            ZeSpace.md,
            ZeSpace.md,
          ),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Align(
                alignment: Alignment.centerLeft,
                child: Text('Menu lainnya', style: ZeText.sectionTitle),
              ),
              const SizedBox(height: ZeSpace.sm),
              for (final entry in const [
                (4, 'Laporan', Icons.bar_chart_outlined),
                (5, 'Transaksi', Icons.receipt_long_outlined),
                (6, 'Pengaturan', Icons.settings_outlined),
              ])
                ListTile(
                  minTileHeight: 52,
                  shape: const RoundedRectangleBorder(
                    borderRadius: ZeRadius.control,
                  ),
                  selected: _activeSection == entry.$1,
                  leading: Icon(entry.$3),
                  title: Text(entry.$2),
                  trailing: const Icon(Icons.chevron_right),
                  onTap: () => Navigator.pop(context, entry.$1),
                ),
            ],
          ),
        ),
      ),
    );
    if (selected != null) _selectSection(selected);
  }

  Widget _buildContent(int section) {
    switch (section) {
      case 0:
        return _DashboardPage(
          database: widget.database,
          onStartCashier: () => _selectSection(1),
        );
      case 1:
        return CashierPage(
          key: ValueKey('cashier_$_scannerQuery'),
          database: widget.database,
          initialSearch: _scannerQuery,
          onSaleCompleted: () => setState(() => _dataVersion++),
        );
      case 2:
        return ProductsPage(database: widget.database);
      case 3:
        return StockPage(database: widget.database);
      case 4:
        return ReportsPage(database: widget.database);
      case 5:
        return TransactionsPage(database: widget.database);
      default:
        return SettingsPage(
          database: widget.database,
          onDatabaseRestored: widget.onDatabaseRestored,
        );
    }
  }
}

enum _StockAlertAction { viewStock }

class _StockNotificationButton extends StatefulWidget {
  const _StockNotificationButton({
    required this.database,
    required this.refreshToken,
    required this.onViewStockAlerts,
  });

  final ZePosDatabase database;
  final int refreshToken;
  final VoidCallback onViewStockAlerts;

  @override
  State<_StockNotificationButton> createState() =>
      _StockNotificationButtonState();
}

class _StockNotificationButtonState extends State<_StockNotificationButton> {
  late Future<DashboardOverview> _overviewFuture;

  @override
  void initState() {
    super.initState();
    _overviewFuture = widget.database.loadDashboardOverview();
  }

  @override
  void didUpdateWidget(covariant _StockNotificationButton oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.refreshToken != widget.refreshToken) {
      _overviewFuture = widget.database.loadDashboardOverview();
    }
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<DashboardOverview>(
      future: _overviewFuture,
      builder: (context, snapshot) {
        if (snapshot.connectionState != ConnectionState.done) {
          return const _StockNotificationIcon(alertCount: 0);
        }
        final overview = snapshot.data;
        final outOfStock = overview?.outOfStockProducts.length ?? 0;
        final lowStock = overview?.lowStockProducts.length ?? 0;
        final alertCount = outOfStock + lowStock;
        final productCount = overview == null
            ? 0
            : overview.outOfStockProducts.length +
                  overview.lowStockProducts.length;
        final summaryHeight = alertCount == 0
            ? 76.0
            : 116.0 +
                  (productCount.clamp(0, 3).toDouble() * 25.0) +
                  (productCount > 3 ? 18.0 : 0);
        return PopupMenuButton<_StockAlertAction>(
          tooltip: alertCount == 0
              ? 'Notifikasi stok'
              : '$alertCount notifikasi stok',
          offset: const Offset(0, 44),
          onSelected: (action) {
            if (action == _StockAlertAction.viewStock) {
              widget.onViewStockAlerts();
            }
          },
          itemBuilder: (context) => [
            PopupMenuItem<_StockAlertAction>(
              enabled: false,
              height: summaryHeight,
              child: SizedBox(
                width: 300,
                child: Padding(
                  padding: const EdgeInsets.symmetric(vertical: 6),
                  child: alertCount == 0
                      ? const _StockAlertMenuContent()
                      : _StockAlertMenuContent(
                          outOfStock: outOfStock,
                          lowStock: lowStock,
                          products: [
                            ...?overview?.outOfStockProducts,
                            ...?overview?.lowStockProducts,
                          ],
                        ),
                ),
              ),
            ),
            const PopupMenuDivider(),
            const PopupMenuItem<_StockAlertAction>(
              value: _StockAlertAction.viewStock,
              child: Row(
                children: [
                  Icon(Icons.inventory_2_outlined, size: 18),
                  SizedBox(width: 10),
                  Text('Lihat stok'),
                ],
              ),
            ),
          ],
          child: _StockNotificationIcon(alertCount: alertCount),
        );
      },
    );
  }
}

class CashierPage extends StatefulWidget {
  const CashierPage({
    super.key,
    required this.database,
    required this.onSaleCompleted,
    this.initialSearch = '',
  });

  final ZePosDatabase database;
  final VoidCallback onSaleCompleted;
  final String initialSearch;

  @override
  State<CashierPage> createState() => _CashierPageState();
}

class _CashierPageState extends State<CashierPage> {
  late String _query;
  String _category = 'Semua';
  late final TextEditingController _searchController;
  final Map<Product, int> _cart = {};
  List<Product> _products = const [];
  Object? _loadingError;
  var _isLoading = true;
  var _isSavingSale = false;

  @override
  void initState() {
    super.initState();
    _query = widget.initialSearch;
    _searchController = TextEditingController(text: _query);
    _loadProducts();
  }

  @override
  void dispose() {
    _searchController.dispose();
    super.dispose();
  }

  Future<void> _loadProducts() async {
    if (mounted) {
      setState(() {
        _isLoading = true;
        _loadingError = null;
      });
    }
    try {
      final products = await widget.database.loadActiveProducts();
      if (!mounted) return;
      setState(() {
        _products = products;
        _isLoading = false;
      });
    } catch (error) {
      if (!mounted) return;
      setState(() {
        _loadingError = error;
        _isLoading = false;
      });
    }
  }

  List<Product> get _visibleProducts => _products.where((product) {
    final matchesCategory =
        _category == 'Semua' || product.category == _category;
    final query = _query.trim().toLowerCase();
    final matchesQuery =
        query.isEmpty ||
        product.name.toLowerCase().contains(query) ||
        product.sku.toLowerCase().contains(query) ||
        product.barcode.contains(query);
    return matchesCategory && matchesQuery;
  }).toList();

  int get _itemCount =>
      _cart.values.fold(0, (total, quantity) => total + quantity);

  int get _total => _cart.entries.fold(
    0,
    (total, entry) => total + (entry.key.price * entry.value),
  );

  void _changeQuantity(Product product, int change) {
    final currentQuantity = _cart[product] ?? 0;
    final nextQuantity = currentQuantity + change;
    if (change > 0 && nextQuantity > product.stock) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Stok ${product.name} tidak mencukupi.')),
      );
      return;
    }
    setState(() {
      if (nextQuantity <= 0) {
        _cart.remove(product);
      } else {
        _cart[product] = nextQuantity;
      }
    });
  }

  Future<void> _openCart() async {
    if (_cart.isEmpty) return;
    await showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: const Color(0xFFEEF2F7),
      builder: (context) => SafeArea(
        child: FractionallySizedBox(
          heightFactor: 0.88,
          child: _CartPanel(
            cart: _cart,
            onChangeQuantity: _changeQuantity,
            onCheckout: _openPayment,
            scrollable: true,
          ),
        ),
      ),
    );
  }

  Future<void> _openPayment() async {
    if (_cart.isEmpty || _isSavingSale) return;
    final totalBeforePayment = _total;
    final paid = await showModalBottomSheet<_PaymentResult>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) => _PaymentSheet(total: totalBeforePayment),
    );

    if (!mounted || paid == null) return;

    setState(() => _isSavingSale = true);
    late SaleReceipt receipt;
    try {
      receipt = await widget.database.completeSale(
        cart: Map<Product, int>.from(_cart),
        paymentMethod: paid.method.storageValue,
        receivedAmount: paid.receivedAmount,
      );
    } on InsufficientStockException catch (error) {
      if (mounted) setState(() => _isSavingSale = false);
      await _loadProducts();
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            'Stok ${error.productName} tinggal ${error.available}. Keranjang perlu diperiksa kembali.',
          ),
        ),
      );
      return;
    } on ProductUnavailableException catch (error) {
      if (mounted) setState(() => _isSavingSale = false);
      await _loadProducts();
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('${error.productName} sudah tidak tersedia.')),
      );
      return;
    } catch (_) {
      if (!mounted) return;
      setState(() => _isSavingSale = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Transaksi belum tersimpan. Silakan coba lagi.'),
        ),
      );
      return;
    }

    if (!mounted) return;
    setState(() {
      _cart.clear();
      _isSavingSale = false;
    });
    await _loadProducts();
    widget.onSaleCompleted();
    if (!mounted) return;
    final printerPreferences = await widget.database.loadPrinterPreferences();
    if (!mounted) return;
    if (printerPreferences.autoPrintReceipt) {
      await _printReceipt(receipt, preferences: printerPreferences);
      if (!mounted) return;
    }
    await showDialog<void>(
      context: context,
      builder: (context) => AlertDialog(
        icon: const Icon(
          Icons.check_circle,
          color: Color(0xFF16A34A),
          size: 44,
        ),
        title: const Text('Pembayaran sudah diterima'),
        content: Text(
          '${receipt.transactionNumber}\n'
          '${paid.method.label} - ${formatRupiah(receipt.total)}'
          '${receipt.changeAmount > 0 ? '\nKembalian: ${formatRupiah(receipt.changeAmount)}' : ''}',
        ),
        actions: [
          OutlinedButton.icon(
            onPressed: () => _printReceipt(receipt),
            icon: const Icon(Icons.print_outlined),
            label: const Text('Cetak Nota'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Transaksi Baru'),
          ),
        ],
      ),
    );
  }

  Future<void> _printReceipt(
    SaleReceipt receipt, {
    PrinterPreferences? preferences,
  }) async {
    try {
      final results = await Future.wait([
        widget.database.loadStoreProfile(),
        widget.database.loadSaleDetail(receipt.saleId),
        widget.database.loadPrintConfiguration(PrintPurpose.receipt),
      ]);
      final profile = results[0] as StoreProfile?;
      if (profile == null) throw StateError('Profil toko belum tersedia.');
      final activePreferences =
          preferences ?? await widget.database.loadPrinterPreferences();
      final printResult = await DocumentExportService.printReceipt(
        profile: profile,
        sale: results[1] as SaleDetail,
        preferences: activePreferences,
        configuration: results[2] as PrintPurposeConfiguration,
      );
      if (!mounted) return;
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text(printResult.userMessage)));
    } on PrintValidationException catch (error) {
      if (!mounted) return;
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text(error.userMessage)));
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Nota belum dapat dicetak. Coba lagi.')),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_isLoading) {
      return const _PageFrame(
        title: 'Kasir',
        subtitle: 'Menyiapkan katalog produk dari perangkat...',
        child: _CashierLoadingState(),
      );
    }
    if (_loadingError != null) {
      return _PageFrame(
        title: 'Kasir',
        subtitle: 'Katalog produk belum tersedia.',
        child: _CashierErrorState(onRetry: _loadProducts),
      );
    }
    return LayoutBuilder(
      builder: (context, constraints) {
        final showCartSideBySide = constraints.maxWidth >= 920;
        final products = _ProductBrowser(
          allProducts: _products,
          products: _visibleProducts,
          query: _query,
          selectedCategory: _category,
          cart: _cart,
          searchController: _searchController,
          onSearchChanged: (value) => setState(() => _query = value),
          onCategoryChanged: (value) => setState(() => _category = value),
          onAdd: (product) => _changeQuantity(product, 1),
        );

        return _PageFrame(
          title: 'Kasir',
          subtitle: 'Cari produk atau scan barcode untuk mulai transaksi.',
          trailing: const _OfflineBadge(),
          child: showCartSideBySide
              ? Row(
                  children: [
                    Expanded(child: products),
                    SizedBox(
                      width: 390,
                      child: _CartPanel(
                        cart: _cart,
                        onChangeQuantity: _changeQuantity,
                        onCheckout: _openPayment,
                      ),
                    ),
                  ],
                )
              : Column(
                  children: [
                    Expanded(child: products),
                    _CartSummaryBar(
                      itemCount: _itemCount,
                      total: _total,
                      onTap: _openCart,
                    ),
                  ],
                ),
        );
      },
    );
  }
}

class _ProductBrowser extends StatelessWidget {
  const _ProductBrowser({
    required this.allProducts,
    required this.products,
    required this.query,
    required this.selectedCategory,
    required this.cart,
    required this.searchController,
    required this.onSearchChanged,
    required this.onCategoryChanged,
    required this.onAdd,
  });

  final List<Product> allProducts;
  final List<Product> products;
  final String query;
  final String selectedCategory;
  final Map<Product, int> cart;
  final TextEditingController searchController;
  final ValueChanged<String> onSearchChanged;
  final ValueChanged<String> onCategoryChanged;
  final ValueChanged<Product> onAdd;

  @override
  Widget build(BuildContext context) {
    final categories = [
      'Semua',
      ...{for (final product in allProducts) product.category},
    ];
    return Padding(
      padding: const EdgeInsets.fromLTRB(24, 4, 16, 16),
      child: Column(
        children: [
          TextField(
            controller: searchController,
            onChanged: onSearchChanged,
            textInputAction: TextInputAction.search,
            decoration: const InputDecoration(
              hintText: 'Cari nama, SKU, atau scan barcode',
              prefixIcon: Icon(Icons.search),
            ),
          ),
          const SizedBox(height: 14),
          SizedBox(
            height: 42,
            child: ListView.separated(
              scrollDirection: Axis.horizontal,
              itemCount: categories.length,
              separatorBuilder: (context, index) => const SizedBox(width: 8),
              itemBuilder: (context, index) {
                final category = categories[index];
                return ChoiceChip(
                  label: Text(category),
                  selected: selectedCategory == category,
                  onSelected: (_) => onCategoryChanged(category),
                  selectedColor: ZeColors.surfaceSelected,
                  labelStyle: TextStyle(
                    color: selectedCategory == category
                        ? ZeColors.primaryStrong
                        : ZeColors.navigationInk,
                    fontWeight: FontWeight.w600,
                  ),
                  shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(10),
                    side: const BorderSide(color: ZeColors.border),
                  ),
                );
              },
            ),
          ),
          const SizedBox(height: 16),
          Expanded(
            child: products.isEmpty
                ? const _EmptyProductState()
                : GridView.builder(
                    gridDelegate:
                        const SliverGridDelegateWithMaxCrossAxisExtent(
                          maxCrossAxisExtent: 220,
                          mainAxisExtent: 172,
                          mainAxisSpacing: 12,
                          crossAxisSpacing: 12,
                        ),
                    itemCount: products.length,
                    itemBuilder: (context, index) {
                      final product = products[index];
                      return _ProductTile(
                        product: product,
                        quantityInCart: cart[product] ?? 0,
                        onAdd: () => onAdd(product),
                      );
                    },
                  ),
          ),
        ],
      ),
    );
  }
}

class _ProductTile extends StatelessWidget {
  const _ProductTile({
    required this.product,
    required this.quantityInCart,
    required this.onAdd,
  });

  final Product product;
  final int quantityInCart;
  final VoidCallback onAdd;

  @override
  Widget build(BuildContext context) {
    final remainingStock = product.stock - quantityInCart;
    return Material(
      color: ZeColors.surface,
      shape: const RoundedRectangleBorder(
        borderRadius: ZeRadius.panel,
        side: BorderSide(color: ZeColors.border),
      ),
      child: InkWell(
        onTap: remainingStock > 0 ? onAdd : null,
        borderRadius: ZeRadius.panel,
        child: Padding(
          padding: const EdgeInsets.all(14),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              if (quantityInCart > 0)
                Align(
                  alignment: Alignment.centerRight,
                  child: Semantics(
                    label: '$quantityInCart item di keranjang',
                    child: Container(
                      constraints: const BoxConstraints(minWidth: 22),

class _CartPanel extends StatelessWidget {
  const _CartPanel({
    required this.cart,
    required this.onChangeQuantity,
    required this.onCheckout,
    this.scrollable = false,
  });

  final Map<Product, int> cart;
  final void Function(Product product, int change) onChangeQuantity;
  final Future<void> Function() onCheckout;
  final bool scrollable;

  int get itemCount =>
      cart.values.fold(0, (total, quantity) => total + quantity);

  int get total => cart.entries.fold(
    0,
    (amount, entry) => amount + (entry.key.price * entry.value),
  );

  @override
  Widget build(BuildContext context) {
    final cartEntries = cart.entries.toList();
    final panel = Padding(
      padding: const EdgeInsets.fromLTRB(16, 4, 24, 16),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              const Text(
                'Keranjang',
                style: TextStyle(fontSize: 18, fontWeight: FontWeight.w600),
              ),
              const SizedBox(width: 8),
              if (itemCount > 0)
                Text(
                  '$itemCount item',
                  style: const TextStyle(color: ZeColors.mutedAlt),
                ),
            ],
          ),
          const SizedBox(height: 12),
          Expanded(
            child: cartEntries.isEmpty
                ? const _EmptyCartState()
                : ListView.separated(
                    itemCount: cartEntries.length,
                    separatorBuilder: (context, index) =>
                        const Divider(height: 1),
                    itemBuilder: (context, index) {
                      final entry = cartEntries[index];
                      return _CartLine(
                        product: entry.key,
                        quantity: entry.value,
                        onChangeQuantity: onChangeQuantity,
                      );
                    },
                  ),
          ),
          const Divider(height: 24),
          _PriceRow(label: 'Subtotal', value: formatRupiah(total)),
          const SizedBox(height: 8),
          const _PriceRow(label: 'Diskon', value: 'Rp0'),
          const Divider(height: 24),
          _PriceRow(
            label: 'Total',
            value: formatRupiah(total),
            emphasize: true,
          ),
          const SizedBox(height: 16),
          SizedBox(
            width: double.infinity,
            child: FilledButton.icon(
              onPressed: cartEntries.isEmpty ? null : onCheckout,
              icon: const Icon(Icons.payments_outlined),
              label: const Text('Bayar'),
            ),
          ),
        ],
      ),
    );

    return Material(
      color: Colors.white,
      borderRadius: scrollable
          ? const BorderRadius.vertical(top: Radius.circular(20))
          : BorderRadius.zero,
      child: panel,
    );
  }
}

  @override

class _PaymentSheetState extends State<_PaymentSheet> {
  late final TextEditingController _cashController;
  PaymentMethod _method = PaymentMethod.cash;
  bool _confirmed = false;

  @override
  void initState() {
    super.initState();
    _cashController = TextEditingController(text: widget.total.toString());
  }

  @override
  void dispose() {
    _cashController.dispose();
    super.dispose();
  }

  int get _cashReceived => int.tryParse(_cashController.text.trim()) ?? 0;

  bool get _canSubmit => _method == PaymentMethod.cash
      ? _cashReceived >= widget.total
      : _confirmed;

  void _submit() {
    if (!_canSubmit) return;
    final receivedAmount = _method == PaymentMethod.cash
        ? _cashReceived
        : widget.total;
    Navigator.pop(
      context,
      _PaymentResult(method: _method, receivedAmount: receivedAmount),
    );
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Padding(
        padding: EdgeInsets.fromLTRB(
          24,
          8,
          24,
          24 + MediaQuery.viewInsetsOf(context).bottom,
        ),
        child: SingleChildScrollView(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text(
                'Pembayaran',
                style: TextStyle(fontSize: 22, fontWeight: FontWeight.w600),
              ),
              const SizedBox(height: 4),
              Text('Total yang harus dibayar: ${formatRupiah(widget.total)}'),
              const SizedBox(height: 20),
              RadioGroup<PaymentMethod>(
                groupValue: _method,
                onChanged: (value) {
                  if (value == null) return;
                  setState(() {
                    _method = value;
                    _confirmed = false;
                  });
                },
                child: Column(
                  children: [
                    for (final method in PaymentMethod.values)
                      RadioListTile<PaymentMethod>(
                        contentPadding: EdgeInsets.zero,
                        value: method,
                        title: Text(method.label),
                        subtitle: Text(method.description),
                        secondary: Icon(method.icon),
                      ),
                  ],
                ),
              ),
              const SizedBox(height: 8),
              if (_method == PaymentMethod.cash) ...[
                TextField(
                  controller: _cashController,
                  keyboardType: TextInputType.number,
                  onChanged: (_) => setState(() {}),
                  decoration: const InputDecoration(
                    labelText: 'Uang diterima',
                    prefixText: 'Rp ',
                  ),
                ),
                const SizedBox(height: 12),
                _PaymentSummary(
                  label: 'Kembalian',
                  value: _cashReceived >= widget.total
                      ? formatRupiah(_cashReceived - widget.total)
                      : 'Nominal belum cukup',
                  isWarning: _cashReceived < widget.total,
                ),
              ] else ...[
                CheckboxListTile(
                  contentPadding: EdgeInsets.zero,
                  value: _confirmed,
                  onChanged: (value) =>
                      setState(() => _confirmed = value ?? false),
                  title: const Text('Pembayaran sudah diterima'),
                  subtitle: const Text(
                    'Pastikan nominal sudah sesuai sebelum melanjutkan.',
                  ),
                  controlAffinity: ListTileControlAffinity.leading,
                ),
              ],
              const SizedBox(height: 20),
              SizedBox(
                width: double.infinity,
                child: FilledButton.icon(
                  onPressed: _canSubmit ? _submit : null,
                  icon: const Icon(Icons.check_circle_outline),
                  label: const Text('Selesaikan Pembayaran'),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class StockPage extends StatefulWidget {
  const StockPage({super.key, required this.database});

  final ZePosDatabase database;

  @override
  State<StockPage> createState() => _StockPageState();
}

class _StockPageState extends State<StockPage> {
  List<Product> _products = const [];
  List<InventoryMovement> _movements = const [];
  String _query = '';
  _StockFilter _filter = _StockFilter.all;
  Object? _loadingError;
  var _isLoading = true;

  @override
  void initState() {
    super.initState();
    _loadStock();
  }

  Future<void> _loadStock() async {
    if (mounted) {
      setState(() {
        _isLoading = true;
        _loadingError = null;
      });
    }
    try {
      final results = await Future.wait([
        widget.database.loadProducts(),
        widget.database.loadInventoryMovements(limit: 12),
      ]);
      if (!mounted) return;
      setState(() {
        _products = results[0] as List<Product>;
        _movements = results[1] as List<InventoryMovement>;
        _isLoading = false;
      });
    } catch (error) {
      if (!mounted) return;
      setState(() {
        _loadingError = error;
        _isLoading = false;
      });
    }
  }

  List<Product> get _visibleProducts {
    final query = _query.trim().toLowerCase();
    return _products.where((product) {
      final matchesQuery =
          query.isEmpty ||
          product.name.toLowerCase().contains(query) ||
          product.sku.toLowerCase().contains(query) ||
          product.barcode.contains(query);
      final matchesFilter = switch (_filter) {
        _StockFilter.all => true,
        _StockFilter.low =>
          product.stock > 0 && product.stock <= product.minimumStock,
        _StockFilter.empty => product.stock == 0,
      };
      return matchesQuery && matchesFilter;
    }).toList();
  }

  Future<void> _openAdjustment([Product? product]) async {
    final saved = await showModalBottomSheet<bool>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) => _StockAdjustmentSheet(
        database: widget.database,
        products: _products,
        initialProduct: product,
      ),
    );
    if (!mounted || saved != true) return;
    await _loadStock();
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(
        content: Text('Stok dan riwayat mutasi telah diperbarui.'),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final lowStock = _products
        .where(
          (product) =>
              product.stock > 0 && product.stock <= product.minimumStock,
        )
        .toList();
    final emptyStock = _products
        .where((product) => product.stock == 0)
        .toList();
    return _PageFrame(
      title: 'Stok',
      subtitle: 'Pantau persediaan dan catat setiap perubahan stok.',
      trailing: FilledButton.icon(
        onPressed: _isLoading || _products.isEmpty ? null : _openAdjustment,
        icon: const Icon(Icons.tune),
        label: const Text('Sesuaikan Stok'),
      ),
      child: _isLoading
          ? const _StockLoadingState()
          : _loadingError != null
          ? _StockErrorState(onRetry: _loadStock)
          : SingleChildScrollView(
              padding: const EdgeInsets.fromLTRB(24, 4, 24, 24),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  _StockMetrics(products: _products),
                  if (lowStock.isNotEmpty || emptyStock.isNotEmpty) ...[
                    const SizedBox(height: 16),
                    _StockAttention(
                      lowStock: lowStock,
                      emptyStock: emptyStock,
                      onFilter: (filter) => setState(() => _filter = filter),
                    ),
                  ],
                  const SizedBox(height: 20),
                  _StockToolbar(
                    filter: _filter,
                    onFilterChanged: (filter) =>
                        setState(() => _filter = filter),
                    onSearchChanged: (query) => setState(() => _query = query),
                  ),
                  const SizedBox(height: 16),
                  LayoutBuilder(
                    builder: (context, constraints) {
                      final catalog = _InventoryCatalog(
                        products: _visibleProducts,
                        onAdjust: _openAdjustment,
                      );
                      final history = _StockHistory(movements: _movements);
                      if (constraints.maxWidth < 1040) {
                        return Column(
                          children: [
                            catalog,
                            const SizedBox(height: 20),
                            history,
                          ],
                        );
                      }
                      return Row(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Expanded(child: catalog),
                          const SizedBox(width: 20),
                          SizedBox(width: 360, child: history),
                        ],
                      );
                    },
                  ),
                ],
              ),
            ),
    );
  }
}

enum _StockChangeType { add, remove }

class _StockAdjustmentSheet extends StatefulWidget {
  const _StockAdjustmentSheet({
    required this.database,
    required this.products,
    this.initialProduct,
  });

  final ZePosDatabase database;
  final List<Product> products;
  final Product? initialProduct;

  @override
  State<_StockAdjustmentSheet> createState() => _StockAdjustmentSheetState();
}

class _StockAdjustmentSheetState extends State<_StockAdjustmentSheet> {
  final _formKey = GlobalKey<FormState>();
  final _quantityController = TextEditingController();
  final _reasonController = TextEditingController();
  late Product _selectedProduct;
  var _type = _StockChangeType.add;
  var _isSaving = false;
  String? _saveError;

  @override
  void initState() {
    super.initState();
    _selectedProduct = widget.initialProduct ?? widget.products.first;
  }

  @override
  void dispose() {
    _quantityController.dispose();
    _reasonController.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() {
      _isSaving = true;
      _saveError = null;
    });
    final quantity = int.parse(_quantityController.text.trim());
    try {
      await widget.database.adjustStock(
        product: _selectedProduct,
        quantityDelta: _type == _StockChangeType.add ? quantity : -quantity,
        reason: _reasonController.text.trim(),
      );
      if (!mounted) return;
      Navigator.pop(context, true);
    } on InsufficientStockException catch (error) {
      if (!mounted) return;
      setState(() {
        _isSaving = false;
        _saveError =
            'Stok ${error.productName} hanya tersedia ${error.available} unit.';
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSaving = false;
        _saveError = 'Perubahan stok belum dapat disimpan. Coba lagi.';
      });
    }
  }

  String? _quantityValidator(String? value) {
    final quantity = int.tryParse(value?.trim() ?? '');
    if (quantity == null || quantity < 1) return 'Masukkan jumlah minimal 1.';
    return null;
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Align(
        alignment: Alignment.bottomCenter,
        heightFactor: 1,
        child: ConstrainedBox(
          constraints: const BoxConstraints(maxWidth: 560),
          child: Padding(
            padding: EdgeInsets.fromLTRB(
              24,
              8,
              24,
              24 + MediaQuery.viewInsetsOf(context).bottom,
            ),
            child: SingleChildScrollView(
              child: Form(
                key: _formKey,
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Text(
                      'Sesuaikan Stok',
                      style: TextStyle(
                        fontSize: 22,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                    const SizedBox(height: 4),
                    const Text(
                      'Perubahan akan langsung tercatat di riwayat stok.',
                      style: TextStyle(color: ZeColors.mutedAlt),
                    ),
                    const SizedBox(height: 20),
                    _FormFieldGroup(
                      label: 'Produk',
                      child: DropdownButtonFormField<String>(
                        initialValue: _selectedProduct.id,
                        isExpanded: true,
                        items: [
                          for (final product in widget.products)
                            DropdownMenuItem(
                              value: product.id,
                              child: Text(
                                '${product.name} — stok ${product.stock}',
                              ),
                            ),
                        ],
                        onChanged: _isSaving
                            ? null
                            : (id) => setState(() {
                                _selectedProduct = widget.products.firstWhere(
                                  (product) => product.id == id,
                                );
                              }),
                      ),
                    ),
                    const SizedBox(height: 16),
                    const Text(
                      'Jenis perubahan',
                      style: TextStyle(fontWeight: FontWeight.w600),
                    ),
                    const SizedBox(height: 8),
                    SegmentedButton<_StockChangeType>(
                      segments: const [
                        ButtonSegment(
                          value: _StockChangeType.add,
                          icon: Icon(Icons.add),
                          label: Text('Tambah'),
                        ),
                        ButtonSegment(
                          value: _StockChangeType.remove,
                          icon: Icon(Icons.remove),
                          label: Text('Kurangi'),
                        ),
                      ],
                      selected: {_type},
                      onSelectionChanged: _isSaving
                          ? null
                          : (value) => setState(() => _type = value.single),
                    ),
                    const SizedBox(height: 16),
                    _FormFieldGroup(
                      label: 'Jumlah unit',
                      child: TextFormField(
                        controller: _quantityController,
                        keyboardType: TextInputType.number,
                        validator: _quantityValidator,
                        decoration: InputDecoration(
                          hintText: _type == _StockChangeType.add
                              ? 'Contoh: 12'
                              : 'Contoh: 2',
                        ),
                      ),
                    ),
                    const SizedBox(height: 16),
                    _FormFieldGroup(
                      label: 'Alasan perubahan',
                      child: TextFormField(
                        controller: _reasonController,
                        textCapitalization: TextCapitalization.sentences,
                        validator: (value) =>
                            value == null || value.trim().isEmpty
                            ? 'Alasan perubahan wajib diisi.'
                            : null,
                        decoration: const InputDecoration(
                          hintText: 'Contoh: Barang datang dari pemasok',
                        ),
                      ),
                    ),
                    if (_saveError case final String message) ...[
                      const SizedBox(height: 12),
                      Text(
                        message,
                        style: const TextStyle(
                          color: ZeColors.danger,
                          fontWeight: FontWeight.w600,
                        ),
                      ),
                    ],
                    const SizedBox(height: 24),
                    Row(
                      children: [
                        Expanded(
                          child: OutlinedButton(
                            onPressed: _isSaving
                                ? null
                                : () => Navigator.pop(context),
                            child: const Text('Batal'),
                          ),
                        ),
                        const SizedBox(width: 12),
                        Expanded(
                          child: FilledButton.icon(
                            onPressed: _isSaving ? null : _save,
                            icon: _isSaving
                                ? const SizedBox(
                                    width: 18,
                                    height: 18,
                                    child: CircularProgressIndicator(
                                      strokeWidth: 2,
                                      color: Colors.white,
                                    ),
                                  )
                                : const Icon(Icons.save_outlined),
                            label: Text(
                              _isSaving ? 'Menyimpan...' : 'Simpan Perubahan',
                            ),
                          ),
                        ),
                      ],
                    ),
                  ],
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }
}

class SettingsPage extends StatefulWidget {
  const SettingsPage({
    super.key,
    required this.database,
    required this.onDatabaseRestored,
  });

  final ZePosDatabase database;
  final VoidCallback onDatabaseRestored;

  @override
  State<SettingsPage> createState() => _SettingsPageState();
}

class _SettingsPageState extends State<SettingsPage> {
  List<BackupRecord> _backups = const [];
  StoreProfile? _profile;
  CredentialInfo? _credentials;
  PrinterPreferences _printerPreferences = const PrinterPreferences();
  Map<PrintPurpose, PrintPurposeConfiguration> _printConfigurations = const {};
  Object? _loadingError;
  var _isLoading = true;
  var _isCreating = false;
  var _isImporting = false;
  var _isResetting = false;
  String? _restoringPath;

  @override
  void initState() {
    super.initState();
    _loadSettings();
  }

  Future<void> _loadSettings() async {
    if (mounted) {
      setState(() {
        _isLoading = true;
        _loadingError = null;
      });
    }
    try {
      final results = await Future.wait([
        widget.database.loadBackups(),
        widget.database.loadStoreProfile(),
        widget.database.loadCredentialInfo(),
        widget.database.loadPrinterPreferences(),
        widget.database.loadPrintConfiguration(PrintPurpose.receipt),
        widget.database.loadPrintConfiguration(PrintPurpose.label),
        widget.database.loadPrintConfiguration(PrintPurpose.report),
      ]);
      if (!mounted) return;
      setState(() {
        _backups = results[0] as List<BackupRecord>;
        _profile = results[1] as StoreProfile?;
        _credentials = results[2] as CredentialInfo?;
        _printerPreferences = results[3] as PrinterPreferences;
        _printConfigurations = {
          PrintPurpose.receipt: results[4] as PrintPurposeConfiguration,
          PrintPurpose.label: results[5] as PrintPurposeConfiguration,
          PrintPurpose.report: results[6] as PrintPurposeConfiguration,
        };
        _isLoading = false;
      });
    } catch (error) {
      if (!mounted) return;
      setState(() {
        _loadingError = error;
        _isLoading = false;
      });
    }
  }

  Future<void> _createBackup() async {
    setState(() => _isCreating = true);
    try {
      await widget.database.createBackup();
      await _loadSettings();
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Backup lokal berhasil dibuat.')),
      );
    } catch (_) {
      if (!mounted) return;
      setState(() => _isCreating = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Backup belum dapat dibuat. Coba lagi.')),
      );
    } finally {
      if (mounted) setState(() => _isCreating = false);
    }
  }

  Future<void> _importBackup() async {
    final selected = await openFile(
      acceptedTypeGroups: const [
        XTypeGroup(label: 'Backup ZePOS', extensions: ['db']),
      ],
    );
    if (!mounted || selected == null) return;
    setState(() => _isImporting = true);
    try {
      final backup = BackupRecord(
        path: selected.path,
        name: selected.name,
        createdAt: DateTime.now(),
        sizeBytes: await selected.length(),
      );
      await _restoreBackup(backup);
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('File backup belum dapat diimpor.')),
      );
    } finally {
      if (mounted) setState(() => _isImporting = false);
    }
  }

  Future<void> _openProfileEditor() async {
    final profile = _profile;
    if (profile == null) return;
    final saved = await showModalBottomSheet<bool>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) =>
          _StoreProfileEditorSheet(database: widget.database, profile: profile),
    );
    if (!mounted || saved != true) return;
    await _loadSettings();
    if (!mounted) return;
    ScaffoldMessenger.of(
      context,
    ).showSnackBar(const SnackBar(content: Text('Profil toko diperbarui.')));
  }

  Future<void> _openCredentialEditor() async {
    final credentials = _credentials;
    if (credentials == null) return;
    final saved = await showModalBottomSheet<bool>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) => _CredentialEditorSheet(
        database: widget.database,
        credentials: credentials,
      ),
    );
    if (!mounted || saved != true) return;
    await _loadSettings();
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Username dan password diperbarui.')),
    );
  }

            ),

        ),
      ),
    );
    confirmationController.dispose();

  }

  Future<void> _resetApplication() async {
    setState(() => _isResetting = true);
    try {
      await widget.database.createBackup();
      await widget.database.clearApplicationData();
      if (!mounted) return;
      widget.onDatabaseRestored();
    } catch (_) {
      if (!mounted) return;
      setState(() => _isResetting = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Reset dibatalkan karena backup belum dapat dibuat.'),
        ),
      );
    }
  }

  Future<void> _restoreBackup(BackupRecord backup) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        icon: const Icon(
          Icons.settings_backup_restore_outlined,
          color: ZeColors.warning,

        title: const Text('Pulihkan backup?'),
        content: Text(
          'Data ZePOS saat ini akan digantikan dengan ${backup.name}. Aplikasi akan dimuat ulang.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Batal'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Pulihkan'),
          ),
        ],
      ),
    );
    if (!mounted || confirmed != true) return;
    setState(() => _restoringPath = backup.path);
    try {
      await widget.database.restoreBackup(backup);
      if (!mounted) return;
      widget.onDatabaseRestored();
    } catch (_) {
      if (!mounted) return;
      setState(() => _restoringPath = null);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(

class _CredentialEditorSheetState extends State<_CredentialEditorSheet> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _usernameController;
  final _currentPasswordController = TextEditingController();
  final _newPasswordController = TextEditingController();
  final _confirmationController = TextEditingController();
  var _isSaving = false;
  String? _saveError;

  @override
  void initState() {
    super.initState();
    _usernameController = TextEditingController(
      text: widget.credentials.username,
    );
  }

  @override
  void dispose() {
    _usernameController.dispose();
    _currentPasswordController.dispose();
    _newPasswordController.dispose();
    _confirmationController.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() {
      _isSaving = true;
      _saveError = null;
    });
    try {
      final verified = await widget.database.verifyCredentials(
        username: widget.credentials.username,
        password: _currentPasswordController.text,
      );
      if (verified == null) {
        if (mounted) {
          setState(() {
            _isSaving = false;
            _saveError = 'Password saat ini tidak sesuai.';
          });
        }
        return;
      }
      await widget.database.saveCredentials(
        username: _usernameController.text.trim(),
        password: _newPasswordController.text,
      );
      if (!mounted) return;
      Navigator.pop(context, true);
    } catch (_) {
      if (mounted) {
        setState(() {
          _isSaving = false;
          _saveError = 'Akses login belum dapat diperbarui. Coba lagi.';
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Align(
        alignment: Alignment.bottomCenter,
        heightFactor: 1,
        child: ConstrainedBox(
          constraints: const BoxConstraints(maxWidth: 560),

            padding: EdgeInsets.fromLTRB(
              24,
              8,
              24,
              24 + MediaQuery.viewInsetsOf(context).bottom,
            ),
            child: SingleChildScrollView(
              child: Form(
                key: _formKey,
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Text(
                      'Ubah akses login',
                      style: TextStyle(
                        fontSize: 22,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                    const SizedBox(height: 4),
                    const Text(
                      'Gunakan password saat ini untuk menyimpan perubahan.',
                      style: TextStyle(color: ZeColors.mutedAlt),
                    ),
                    const SizedBox(height: 20),
                    _FormFieldGroup(
                      label: 'Username',
                      child: TextFormField(
                        controller: _usernameController,
                        autocorrect: false,
                        validator: (value) =>
                            value == null || value.trim().isEmpty
                            ? 'Username wajib diisi.'
                            : null,
                      ),
                    ),
                    const SizedBox(height: 14),
                    _FormFieldGroup(
                      label: 'Password saat ini',
                      child: TextFormField(
                        controller: _currentPasswordController,
                        obscureText: true,
                        enableSuggestions: false,
                        autocorrect: false,
                        validator: (value) => value == null || value.isEmpty
                            ? 'Masukkan password saat ini.'
                            : null,
                      ),
                    ),
                    const SizedBox(height: 14),
                    _FormFieldGroup(
                      label: 'Password baru',
                      child: TextFormField(
                        controller: _newPasswordController,
                        obscureText: true,
                        enableSuggestions: false,
                        autocorrect: false,
                        validator: (value) => value == null || value.length < 6
                            ? 'Password minimal 6 karakter.'
                            : null,
                      ),
                    ),
                    const SizedBox(height: 14),
                    _FormFieldGroup(
                      label: 'Ulangi password baru',
                      child: TextFormField(

                        enableSuggestions: false,
                        autocorrect: false,

                            value != _newPasswordController.text

                                child: CircularProgressIndicator(

                              )

class _UniversalPrinterSettingsSheet extends StatefulWidget {
  const _UniversalPrinterSettingsSheet({
    required this.database,
    required this.configurations,
    this.initialPurpose = PrintPurpose.receipt,
  });

  final ZePosDatabase database;
  final Map<PrintPurpose, PrintPurposeConfiguration> configurations;
  final PrintPurpose initialPurpose;

  @override
  State<_UniversalPrinterSettingsSheet> createState() =>
      _UniversalPrinterSettingsSheetState();
}

class _UniversalPrinterSettingsSheetState
    extends State<_UniversalPrinterSettingsSheet> {
  late final Map<PrintPurpose, _PrintConfigurationDraft> _drafts;
  List<Printer> _printers = const [];
  late PrintPurpose _purpose;
  bool _isDiscovering = true;
  bool _isSaving = false;
  bool _isTesting = false;
  String? _message;

  @override
  void initState() {
    super.initState();
    _purpose = widget.initialPurpose;
    _drafts = {
      for (final purpose in PrintPurpose.values)
        purpose: _PrintConfigurationDraft.fromConfiguration(
          widget.configurations[purpose] ??
              PrintPurposeConfiguration(
                profile: BuiltInPrinterProfiles.forPurpose(purpose).first,
                media: BuiltInPrinterProfiles.forPurpose(
                  purpose,
                ).first.defaultMedia,
                policy: PrintPolicy(purpose: purpose),
              ),
        ),
    };
    _discover();
  }

  Future<void> _discover() async {
    setState(() {
      _isDiscovering = true;
      _message = null;
    });
    try {
      final printers = await UniversalPrintService.discoverPrinters();
      if (!mounted) return;
      setState(() {
        _printers = printers;
        _isDiscovering = false;
        if (printers.isEmpty) {
          _message = 'Windows belum melaporkan printer yang tersedia.';
        }
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isDiscovering = false;
        _message = 'Daftar printer Windows belum dapat dibaca.';
      });
    }
  }

  Future<void> _test() async {
    final draft = _drafts[_purpose]!;
    setState(() {
      _isTesting = true;
      _message = null;
    });
    PrintJobResult result;
    try {
      result = _purpose == PrintPurpose.label
          ? await DocumentExportService.printLabelCalibration(
              configuration: draft.configuration,
              copies: 10,
            )
          : await UniversalPrintService.testPrinter(
              profile: draft.profile,
              media: draft.media,
              policy: draft.policy,
              binding: draft.binding,
            );
    } on PrintValidationException catch (error) {
      result = PrintJobResult(
        jobId: 'validation',
        state: PrintJobState.failedBeforeSpool,
        userMessage: error.userMessage,
        errorCode: 'INVALID_LABEL_CONFIGURATION',
        retrySafe: true,
      );
    }
    if (!mounted) return;
    setState(() {
      _isTesting = false;
      _message = result.userMessage;
    });
  }

  Future<void> _save() async {
    for (final draft in _drafts.values) {
      if (draft.media.widthMm <= 0 ||
          (draft.purpose == PrintPurpose.label &&
              (draft.media.heightMm ?? 0) <= 0)) {
        setState(() => _message = 'Ukuran media harus lebih besar dari nol.');
        return;
      }
      if (draft.media.widthMm > draft.profile.maxWidthMm) {
        setState(
          () => _message =
              'Lebar media ${draft.media.name} melebihi kemampuan profil printer.',
        );
        return;
      }
      if (draft.purpose == PrintPurpose.label) {
        final errors = LabelPrintValidator.validate(
          media: draft.media,
          profile: draft.profile,
          policy: draft.policy,
          barcode: 'ZEPOS-VALIDATION',
          copies: 1,
          binding: draft.binding,
        );
        if (errors.isNotEmpty) {
          setState(() => _message = errors.first);
          return;
        }
      }
    }
    setState(() {
      _isSaving = true;
      _message = null;
    });
    try {
      for (final draft in _drafts.values) {
        await widget.database.savePrintConfiguration(draft.configuration);
      }
      if (!mounted) return;
      Navigator.pop(context, true);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSaving = false;
        _message = 'Konfigurasi printer belum dapat disimpan.';
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    final draft = _drafts[_purpose]!;
    final isAndroid = Theme.of(context).platform == TargetPlatform.android;
    final profiles = BuiltInPrinterProfiles.forPurpose(_purpose);
    final visiblePrinters = [..._printers];
    if (draft.binding case final PrinterBinding binding) {
      if (!visiblePrinters.any(
        (printer) => printer.url == binding.printerUrl,
      )) {
        visiblePrinters.insert(
          0,
          Printer(
            url: binding.printerUrl,
            name: '${binding.printerName} · Tidak ditemukan',
            model: binding.printerModel,
            isAvailable: false,
          ),
        );
      }
    }
    return SafeArea(
      child: Align(
        alignment: Alignment.bottomCenter,
        heightFactor: 1,
        child: ConstrainedBox(
          constraints: BoxConstraints(
            maxWidth: 820,
            maxHeight: MediaQuery.sizeOf(context).height * .9,
          ),
          child: SingleChildScrollView(
            padding: EdgeInsets.fromLTRB(
              ZeSpace.lg,
              ZeSpace.xs,
              ZeSpace.lg,
              ZeSpace.lg + MediaQuery.viewInsetsOf(context).bottom,
            ),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const Text('Printer universal', style: ZeText.sectionTitle),
                const SizedBox(height: ZeSpace.xxs),
                Text(
                  isAndroid
                      ? 'Pilih printer Bluetooth yang sudah dipasangkan untuk setiap jenis dokumen.'
                      : 'Pilih printer Windows dan profil kemampuan untuk setiap jenis dokumen.',
                  style: ZeText.meta,
                ),
                if (isAndroid) ...[
                  const SizedBox(height: ZeSpace.sm),
                  Container(
                    width: double.infinity,
                    padding: const EdgeInsets.all(ZeSpace.sm),
                    decoration: BoxDecoration(
                      color: ZeColors.primarySoft,
                      borderRadius: ZeRadius.control,
                    ),
                    child: const Row(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Icon(
                          Icons.bluetooth_outlined,
                          color: ZeColors.primaryStrong,
                        ),
                        SizedBox(width: ZeSpace.xs),
                        Expanded(
                          child: Text(
                            'Pasangkan printer terlebih dahulu dari Pengaturan Bluetooth Android, lalu tekan muat ulang.',
                            style: ZeText.body,
                          ),
                        ),

class _PrintConfigurationDraft {
  _PrintConfigurationDraft({
    required this.purpose,
    required this.profile,
    required this.media,
    required this.dispatchMode,
    required this.copies,
    required this.fallbackToDialog,
    required this.usePrinterSettings,
    required this.openCashDrawer,
    this.binding,
  });

  factory _PrintConfigurationDraft.fromConfiguration(
    PrintPurposeConfiguration configuration,
  ) => _PrintConfigurationDraft(
    purpose: configuration.policy.purpose,
    profile: configuration.profile,
    media: configuration.media,
    dispatchMode: configuration.policy.dispatchMode,
    copies: configuration.policy.copies,
    fallbackToDialog: configuration.policy.fallbackToDialog,
    usePrinterSettings: configuration.policy.usePrinterSettings,
    openCashDrawer: configuration.policy.openCashDrawer,
    binding: configuration.binding,
  );

  final PrintPurpose purpose;
  PrinterProfile profile;
  MediaPreset media;
  PrintDispatchMode dispatchMode;
  int copies;
  bool fallbackToDialog;
  bool usePrinterSettings;
  bool openCashDrawer;
  PrinterBinding? binding;

  PrintPolicy get policy => PrintPolicy(
    purpose: purpose,
    dispatchMode: dispatchMode,
    copies: copies,
    fallbackToDialog: fallbackToDialog,
    usePrinterSettings: usePrinterSettings,
    openCashDrawer: openCashDrawer,
  );

  PrintPurposeConfiguration get configuration => PrintPurposeConfiguration(
    profile: profile,
    media: media,
    policy: policy,
    binding: binding == null
        ? null
        : PrinterBinding(
            id: binding!.id,
            purpose: purpose,
            profileId: profile.id,
            printerUrl: binding!.printerUrl,
            printerName: binding!.printerName,
            printerModel: binding!.printerModel,
            updatedAt: DateTime.now(),
          ),
  );
}

class _MediaPresetEditor extends StatelessWidget {
  const _MediaPresetEditor({
    super.key,
    required this.draft,
    required this.enabled,
    required this.onChanged,
  });

  final _PrintConfigurationDraft draft;
  final bool enabled;
  final VoidCallback onChanged;

  double? _number(String value) =>
      double.tryParse(value.trim().replaceAll(',', '.'));

  void _update({
    double? width,
    double? height,
    double? gap,
    int? columns,
    int? rotation,
    double? safeMargin,
    double? offsetX,
    double? offsetY,
  }) {
    final current = draft.media;
    draft.media = MediaPreset(
      id: 'custom-${draft.purpose.name}',
      purpose: draft.purpose,
      name: draft.purpose == PrintPurpose.label
          ? 'Media label kustom'
          : draft.purpose == PrintPurpose.receipt
          ? 'Thermal ${width ?? current.widthMm} mm'
          : 'Dokumen kustom',
      widthMm: width ?? current.widthMm,
      heightMm: height ?? current.heightMm,
      gapMm: gap ?? current.gapMm,
      columns: columns ?? current.columns,
      rotation: rotation ?? current.rotation,
      safeMarginMm: safeMargin ?? current.safeMarginMm,
      offsetXmm: offsetX ?? current.offsetXmm,
      offsetYmm: offsetY ?? current.offsetYmm,
      sensorMode: current.sensorMode,
    );
    onChanged();
  }

  @override
  Widget build(BuildContext context) {
    final media = draft.media;
    final isLabel = draft.purpose == PrintPurpose.label;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        const Text('Media dan kalibrasi', style: ZeText.cardTitle),
        const SizedBox(height: ZeSpace.xxs),
        Text(
          isLabel
              ? 'Pitch halaman selalu tinggi label + GAP agar posisi tidak bergeser.'
              : 'Ukuran fisik PDF mengikuti media printer yang dipilih.',
          style: ZeText.meta,
        ),
        const SizedBox(height: ZeSpace.md),
        Wrap(
          spacing: ZeSpace.sm,
          runSpacing: ZeSpace.sm,
          children: [
            SizedBox(
              width: 150,
              child: TextFormField(
                initialValue: media.widthMm.toStringAsFixed(1),
                enabled: enabled,
                keyboardType: const TextInputType.numberWithOptions(
                  decimal: true,
                ),
                decoration: const InputDecoration(labelText: 'Lebar (mm)'),
                onChanged: (value) {
                  final parsed = _number(value);
                  if (parsed != null) _update(width: parsed);
                },
              ),
            ),
            if (isLabel || draft.purpose == PrintPurpose.report)
              SizedBox(
                width: 150,
                child: TextFormField(
                  initialValue: media.heightMm?.toStringAsFixed(1) ?? '',
                  enabled: enabled,
                  keyboardType: const TextInputType.numberWithOptions(
                    decimal: true,
                  ),
                  decoration: const InputDecoration(labelText: 'Tinggi (mm)'),
                  onChanged: (value) {
                    final parsed = _number(value);
                    if (parsed != null) _update(height: parsed);
                  },
                ),
              ),
            if (isLabel) ...[
              SizedBox(
                width: 130,
                child: TextFormField(
                  initialValue: media.gapMm.toStringAsFixed(1),
                  enabled: enabled,
                  keyboardType: const TextInputType.numberWithOptions(
                    decimal: true,
                  ),
                  decoration: const InputDecoration(labelText: 'GAP (mm)'),
                  onChanged: (value) {
                    final parsed = _number(value);
                    if (parsed != null) _update(gap: parsed);
                  },
                ),
              ),
              SizedBox(
                width: 130,
                child: DropdownButtonFormField<int>(
                  initialValue: media.columns.clamp(1, 3),
                  decoration: const InputDecoration(labelText: 'Kolom'),
                  items: [
                    for (var column = 1; column <= 3; column++)
                      DropdownMenuItem(
                        value: column,
                        child: Text('$column kolom'),
                      ),
                  ],
                  onChanged: enabled
                      ? (value) {
                          if (value != null) _update(columns: value);
                        }
                      : null,
                ),
              ),
              SizedBox(
                width: 130,
                child: DropdownButtonFormField<int>(
                  initialValue: media.rotation,
                  decoration: const InputDecoration(labelText: 'Rotasi'),
                  items: const [
                    DropdownMenuItem(value: 0, child: Text('0°')),
                    DropdownMenuItem(value: 90, child: Text('90°')),
                    DropdownMenuItem(value: 180, child: Text('180°')),
                    DropdownMenuItem(value: 270, child: Text('270°')),
                  ],
                  onChanged: enabled
                      ? (value) {
                          if (value != null) _update(rotation: value);
                        }
                      : null,
                ),
              ),
            ],
            SizedBox(
              width: 150,
              child: TextFormField(
                initialValue: media.safeMarginMm.toStringAsFixed(1),
                enabled: enabled,
                keyboardType: const TextInputType.numberWithOptions(
                  decimal: true,
                ),
                decoration: const InputDecoration(labelText: 'Margin aman'),
                onChanged: (value) {
                  final parsed = _number(value);
                  if (parsed != null) _update(safeMargin: parsed);
                },
              ),
            ),
            if (isLabel) ...[
              SizedBox(
                width: 130,
                child: TextFormField(
                  initialValue: media.offsetXmm.toStringAsFixed(1),
                  enabled: enabled,
                  keyboardType: const TextInputType.numberWithOptions(
                    decimal: true,
                    signed: true,
                  ),
                  decoration: const InputDecoration(labelText: 'Offset X'),
                  onChanged: (value) {
                    final parsed = _number(value);
                    if (parsed != null) _update(offsetX: parsed);
                  },
                ),
              ),
              SizedBox(
                width: 130,
                child: TextFormField(
                  initialValue: media.offsetYmm.toStringAsFixed(1),
                  enabled: enabled,
                  keyboardType: const TextInputType.numberWithOptions(
                    decimal: true,
                    signed: true,
                  ),
                  decoration: const InputDecoration(labelText: 'Offset Y'),
                  onChanged: (value) {
                    final parsed = _number(value);
                    if (parsed != null) _update(offsetY: parsed);
                  },
                ),
              ),
            ],
          ],
        ),
        if (isLabel) ...[
          const SizedBox(height: ZeSpace.sm),
          Text(
            'Pitch keluaran: ${media.widthMm.toStringAsFixed(1)} × ${(media.pitchHeightMm ?? 0).toStringAsFixed(1)} mm',
            style: ZeText.meta,
          ),
        ],
      ],
    );
  }
}

  late bool _autoPrint;
  late bool _showStoreAddress;
  late bool _showStorePhone;
  late bool _showTransactionNumber;
  late bool _showTransactionTime;
  late bool _showPaymentMethod;
  late bool _showPaymentDetails;
  late final TextEditingController _footerController;
  var _isSaving = false;

  @override
  void initState() {
    super.initState();
    _paperWidth = widget.preferences.receiptPaperWidth == 58 ? 58 : 80;
    _autoPrint = widget.preferences.autoPrintReceipt;
    _showStoreAddress = widget.preferences.showStoreAddress;
    _showStorePhone = widget.preferences.showStorePhone;
    _showTransactionNumber = widget.preferences.showTransactionNumber;
    _showTransactionTime = widget.preferences.showTransactionTime;
    _showPaymentMethod = widget.preferences.showPaymentMethod;
    _showPaymentDetails = widget.preferences.showPaymentDetails;
    _footerController = TextEditingController(
      text: widget.preferences.footerMessage,
    );
  }

  void dispose() {
    _footerController.dispose();
    super.dispose();
  }

    setState(() => _isSaving = true);
    try {
      await widget.database.savePrinterPreferences(
        PrinterPreferences(
          receiptPaperWidth: _paperWidth,
          autoPrintReceipt: _autoPrint,
          showStoreAddress: _showStoreAddress,
          showStorePhone: _showStorePhone,
          showTransactionNumber: _showTransactionNumber,
          showTransactionTime: _showTransactionTime,
          showPaymentMethod: _showPaymentMethod,
          showPaymentDetails: _showPaymentDetails,
          footerMessage: _footerController.text.trim(),
        ),
      );
      final printConfiguration = await widget.database.loadPrintConfiguration(
        PrintPurpose.receipt,
      );
      await widget.database.savePrintConfiguration(
        PrintPurposeConfiguration(
          profile: printConfiguration.profile,
          media: _paperWidth == 58
              ? BuiltInPrinterProfiles.receipt58
              : BuiltInPrinterProfiles.receipt80,
          policy: printConfiguration.policy,
          binding: printConfiguration.binding,
        ),
      );

      if (mounted) setState(() => _isSaving = false);

class ReportsPage extends StatefulWidget {
  const ReportsPage({super.key, required this.database});

  final ZePosDatabase database;

  @override
  State<ReportsPage> createState() => _ReportsPageState();
}

class _ReportsPageState extends State<ReportsPage> {
  late DateTime _start;
  late DateTime _end;
  Future<SalesReport>? _reportFuture;
  Future<_ClosingOverview>? _closingFuture;
  var _isExportingPdf = false;
  var _isExportingExcel = false;
  var _isPrintingReport = false;
  var _isClosingToday = false;

  @override
  void initState() {
    super.initState();
    final now = DateTime.now();
    _start = DateTime(now.year, now.month, now.day);
    _end = _start;
    _loadReport();
  }

  void _loadReport() {
    setState(() {
      _reportFuture = widget.database.loadSalesReport(start: _start, end: _end);
      _closingFuture = _loadClosingOverview();
    });
  }

  Future<_ClosingOverview> _loadClosingOverview() async {
    final today = DateTime.now();
    final values = await Future.wait([
      widget.database.loadDailyClosing(today),
      widget.database.loadMonthlyClosings(DateTime(today.year, today.month)),
    ]);
    return _ClosingOverview(
      today: values[0] as DailyClosing?,
      month: values[1] as List<DailyClosing>,
    );
  }

  Future<void> _selectPeriod() async {
    final range = await showDateRangePicker(
      context: context,
      firstDate: DateTime(2020),
      lastDate: DateTime.now(),
      initialDateRange: DateTimeRange(start: _start, end: _end),
      helpText: 'Pilih periode laporan',
      saveText: 'Terapkan',
    );
    if (range == null || !mounted) return;
    setState(() {
      _start = range.start;
      _end = range.end;
      _reportFuture = widget.database.loadSalesReport(start: _start, end: _end);
      _closingFuture = _loadClosingOverview();
    });
  }

  Future<void> _closeToday() async {
    setState(() => _isClosingToday = true);
    try {
      await widget.database.closeDailySales(DateTime.now());
      if (!mounted) return;
      _loadReport();
      setState(() => _isClosingToday = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Closing hari ini berhasil disimpan.')),
      );
    } catch (_) {
      if (!mounted) return;
      setState(() => _isClosingToday = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Closing harian belum dapat disimpan.')),
      );
    }
  }

  Future<void> _exportPdf(SalesReport report) async {
    setState(() => _isExportingPdf = true);
    try {
      final profile = await widget.database.loadStoreProfile();
      final saved = await DocumentExportService.saveSalesReportPdf(
        profile: profile,
        start: _start,
        end: _end,
        report: report,
      );
      if (!mounted || !saved) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Laporan PDF berhasil disimpan.')),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Laporan PDF belum dapat disimpan.')),
      );
    } finally {
      if (mounted) setState(() => _isExportingPdf = false);
    }
  }

  Future<void> _exportExcel(SalesReport report) async {
    setState(() => _isExportingExcel = true);
    try {
      final profile = await widget.database.loadStoreProfile();
      final saved = await DocumentExportService.saveSalesReportExcel(
        profile: profile,
        start: _start,
        end: _end,
        report: report,
      );
      if (!mounted || !saved) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Laporan Excel berhasil disimpan.')),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Laporan Excel belum dapat disimpan.')),
      );
    } finally {
      if (mounted) setState(() => _isExportingExcel = false);
    }
  }

  Future<void> _printReport(SalesReport report) async {
    setState(() => _isPrintingReport = true);
    try {
      final values = await Future.wait([
        widget.database.loadStoreProfile(),
        widget.database.loadPrintConfiguration(PrintPurpose.report),
      ]);
      final result = await DocumentExportService.printSalesReport(
        profile: values[0] as StoreProfile?,
        start: _start,
        end: _end,
        report: report,
        configuration: values[1] as PrintPurposeConfiguration,
      );
      if (!mounted) return;
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text(result.userMessage)));
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Laporan belum dapat dicetak.')),
      );
    } finally {
      if (mounted) setState(() => _isPrintingReport = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return _PageFrame(
      title: 'Laporan',
      subtitle: formatReportPeriod(_start, _end),
      trailing: OutlinedButton.icon(
        onPressed: _selectPeriod,
        icon: const Icon(Icons.date_range_outlined),
        label: const Text('Pilih Periode'),
      ),
      child: FutureBuilder<SalesReport>(
        future: _reportFuture,
        builder: (context, snapshot) {
          if (snapshot.connectionState != ConnectionState.done) {
            return const _ReportsLoadingState();
          }
          if (snapshot.hasError || !snapshot.hasData) {
            return _ReportsErrorState(onRetry: _loadReport);
          }
          final report = snapshot.data!;
          return SingleChildScrollView(
            padding: ZeSpace.page,
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                _ReportMetrics(report: report),
                const SizedBox(height: ZeSpace.md),
                Align(
                  alignment: Alignment.centerLeft,
                  child: Wrap(
                    spacing: ZeSpace.xs,
                    runSpacing: ZeSpace.xs,
                    children: [
                      OutlinedButton.icon(
                        onPressed: _isExportingPdf
                            ? null
                            : () => _exportPdf(report),
                        icon: _isExportingPdf
                            ? const SizedBox(
                                width: 16,
                                height: 16,
                                child: CircularProgressIndicator(
                                  strokeWidth: 2,
                                ),
                              )
                            : const Icon(Icons.picture_as_pdf_outlined),
                        label: Text(
                          _isExportingPdf ? 'Menyiapkan PDF...' : 'Ekspor PDF',
                        ),
                      ),
                      OutlinedButton.icon(
                        onPressed: _isExportingExcel
                            ? null
                            : () => _exportExcel(report),
                        icon: _isExportingExcel
                            ? const SizedBox(
                                width: 16,
                                height: 16,
                                child: CircularProgressIndicator(
                                  strokeWidth: 2,
                                ),
                              )
                            : const Icon(Icons.table_view_outlined),
                        label: Text(
                          _isExportingExcel
                              ? 'Menyiapkan Excel...'
                              : 'Ekspor Excel',
                        ),
                      ),
                      FilledButton.icon(
                        onPressed: _isPrintingReport
                            ? null
                            : () => _printReport(report),
                        icon: _isPrintingReport
                            ? const SizedBox(
                                width: 16,
                                height: 16,
                                child: CircularProgressIndicator(
                                  strokeWidth: 2,
                                  color: Colors.white,
                                ),
                              )
                            : const Icon(Icons.print_outlined),
                        label: Text(
                          _isPrintingReport ? 'Mengirim...' : 'Cetak Laporan',
                        ),
                      ),
                    ],
                  ),
                ),
                const SizedBox(height: ZeSpace.lg),
                _BentoPair(
                  primary: _PaymentBreakdownPanel(report: report),
                  secondary: _TopProductsPanel(report: report),
                ),
                const SizedBox(height: ZeSpace.lg),
                FutureBuilder<_ClosingOverview>(
                  future: _closingFuture,
                  builder: (context, closingSnapshot) {
                    if (closingSnapshot.connectionState !=
                            ConnectionState.done ||
                        !closingSnapshot.hasData) {
                      return const _LoadingPanel(height: 150);
                    }
                    return _ClosingPanel(
                      overview: closingSnapshot.data!,
                      isClosing: _isClosingToday,
                      onCloseToday: _closeToday,
                    );
                  },
                ),
              ],
            ),
          );
        },
      ),
    );
  }
}

class _ClosingPanel extends StatelessWidget {
  const _ClosingPanel({
    required this.overview,
    required this.isClosing,
    required this.onCloseToday,
  });

  final _ClosingOverview overview;
  final bool isClosing;
  final Future<void> Function() onCloseToday;

  @override
  Widget build(BuildContext context) {
    final today = overview.today;
    final monthlySales = overview.month.fold<int>(
      0,
      (total, closing) => total + closing.salesTotal,
    );
    return _ReportPanel(
      title: 'Closing & Rekap Bulanan',
      subtitle:
          'Hari sebelumnya yang belum ditutup akan dicatat otomatis saat ZePOS dibuka.',
      child: LayoutBuilder(
        builder: (context, _) {
          final daily = _ClosingDetail(
            label: today == null ? 'Belum ditutup' : 'Closing tersimpan',
            value: today == null
                ? 'Tutup penjualan hari ini saat operasional selesai.'
                : '${today.transactionCount} transaksi · ${formatRupiah(today.salesTotal)}',
            icon: today == null
                ? Icons.lock_clock_outlined
                : Icons.task_alt_outlined,
            color: today == null ? ZeColors.warning : ZeColors.success,
            action: FilledButton.icon(
              onPressed: isClosing ? null : onCloseToday,
              icon: isClosing
                  ? const SizedBox(
                      width: 16,
                      height: 16,
                      child: CircularProgressIndicator(
                        strokeWidth: 2,
                        color: Colors.white,
                      ),
                    )
                  : const Icon(Icons.lock_outline),
              label: Text(isClosing ? 'Menyimpan...' : 'Closing Hari Ini'),
            ),
          );
          final monthly = _ClosingDetail(
            label: 'Rekap bulan ini',

class TransactionsPage extends StatefulWidget {
  const TransactionsPage({super.key, required this.database});

  final ZePosDatabase database;

  @override
  State<TransactionsPage> createState() => _TransactionsPageState();
}

class _TransactionsPageState extends State<TransactionsPage> {
  List<SaleRecord> _sales = const [];
  String _query = '';
  Object? _loadingError;
  var _isLoading = true;

  @override
  void initState() {
    super.initState();
    _loadSales();
  }

  Future<void> _loadSales() async {
    if (mounted) {
      setState(() {
        _isLoading = true;
        _loadingError = null;
      });
    }
    try {
      final sales = await widget.database.loadSales();
      if (!mounted) return;
      setState(() {
        _sales = sales;
        _isLoading = false;
      });
    } catch (error) {
      if (!mounted) return;
      setState(() {
        _loadingError = error;
        _isLoading = false;
      });
    }
  }

  List<SaleRecord> get _visibleSales {
    final query = _query.trim().toLowerCase();
    if (query.isEmpty) return _sales;
    return _sales.where((sale) {
      return sale.transactionNumber.toLowerCase().contains(query) ||
          paymentMethodLabel(
            sale.paymentMethod,
          ).toLowerCase().contains(query) ||
          transactionStatusLabel(sale.status).toLowerCase().contains(query);
    }).toList();
  }

  Future<void> _openDetail(SaleRecord sale) async {
    final changed = await showModalBottomSheet<bool>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) =>
          _TransactionDetailSheet(database: widget.database, sale: sale),
    );
    if (!mounted || changed != true) return;
    await _loadSales();
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Transaksi dan stok berhasil diperbarui.')),
    );
  }

  @override
  Widget build(BuildContext context) {
    return _PageFrame(
      title: 'Transaksi',
      subtitle: 'Lihat detail penjualan, pembatalan, dan retur.',
      child: _isLoading
          ? const _TransactionsLoadingState()
          : _loadingError != null
          ? _TransactionsErrorState(onRetry: _loadSales)
          : Padding(
              padding: ZeSpace.page,
              child: Column(
                children: [
                  TextField(
                    onChanged: (query) => setState(() => _query = query),
                    decoration: const InputDecoration(
                      hintText: 'Cari nomor transaksi, pembayaran, atau status',
                      prefixIcon: Icon(Icons.search),
                    ),
                  ),
                  const SizedBox(height: 16),
                  Expanded(
                    child: _TransactionsList(
                      sales: _visibleSales,
                      onOpen: _openDetail,
                    ),
                  ),
                ],
              ),
            ),
    );
  }
}

class _TransactionDetailSheet extends StatefulWidget {
  const _TransactionDetailSheet({required this.database, required this.sale});

  final ZePosDatabase database;
  final SaleRecord sale;

  @override
  State<_TransactionDetailSheet> createState() =>
      _TransactionDetailSheetState();
}

class _TransactionDetailSheetState extends State<_TransactionDetailSheet> {
  late Future<SaleDetail> _detailFuture;
  var _isSubmitting = false;

  @override
  void initState() {
    super.initState();
    _detailFuture = widget.database.loadSaleDetail(widget.sale.id);
  }

  Future<void> _reverse({required bool isReturn}) async {
    final action = isReturn ? 'Retur Penuh' : 'Batalkan Transaksi';
    final message = isReturn
        ? 'Seluruh item akan dikembalikan ke stok dan transaksi tidak dapat diubah lagi.'
        : 'Seluruh item akan dikembalikan ke stok. Gunakan untuk transaksi yang dibatalkan.';
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        icon: Icon(
          isReturn ? Icons.assignment_return_outlined : Icons.cancel_outlined,
          color: ZeColors.danger,
        ),
        title: Text('$action?'),
        content: Text(message),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Kembali'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            style: FilledButton.styleFrom(backgroundColor: ZeColors.danger),
            child: Text(action),
          ),
        ],
      ),
    );
    if (!mounted || confirmed != true) return;
    setState(() => _isSubmitting = true);
    try {
      if (isReturn) {
        await widget.database.returnSale(widget.sale);
      } else {
        await widget.database.voidSale(widget.sale);
      }
      if (!mounted) return;
      Navigator.pop(context, true);
    } on SaleAlreadyReversedException {
      if (!mounted) return;
      setState(() => _isSubmitting = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Transaksi ini sudah tidak dapat diubah.'),
        ),
      );
    } catch (_) {
      if (!mounted) return;
      setState(() => _isSubmitting = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Transaksi belum dapat diperbarui. Coba lagi.'),
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: ConstrainedBox(
        constraints: BoxConstraints(
          maxHeight: MediaQuery.sizeOf(context).height * .88,
          maxWidth: 640,
        ),
        child: FutureBuilder<SaleDetail>(
          future: _detailFuture,
          builder: (context, snapshot) {
            if (snapshot.connectionState != ConnectionState.done) {
              return const Center(
                child: Padding(
                  padding: EdgeInsets.all(48),
                  child: CircularProgressIndicator(),
                ),
              );
            }
            if (snapshot.hasError || !snapshot.hasData) {
              return const Center(
                child: Padding(
                  padding: EdgeInsets.all(48),
                  child: Text('Detail transaksi belum dapat dimuat.'),
                ),
              );
            }
            final detail = snapshot.data!;
            return Padding(
              padding: const EdgeInsets.fromLTRB(24, 4, 24, 24),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    detail.sale.transactionNumber,
                    style: const TextStyle(
                      fontSize: 22,
                      fontWeight: FontWeight.w600,
                    ),
                  ),
                  const SizedBox(height: 4),
                  Text(
                    formatDateTime(detail.sale.createdAt),
                    style: const TextStyle(color: ZeColors.mutedAlt),
                  ),
                  const SizedBox(height: 12),
                  _TransactionStatusBadge(status: detail.sale.status),
                  const SizedBox(height: 20),
                  const Text(
                    'Item Penjualan',
                    style: TextStyle(fontSize: 17, fontWeight: FontWeight.w600),
                  ),
                  const SizedBox(height: 8),
                  Expanded(
                    child: ListView.separated(
                      itemCount: detail.items.length,
                      separatorBuilder: (context, index) =>
                          const Divider(height: 16),
                      itemBuilder: (context, index) {
                        final item = detail.items[index];
                        return Row(
                          children: [
                            Expanded(
                              child: Column(

class ProductsPage extends StatefulWidget {
  const ProductsPage({super.key, required this.database});

  final ZePosDatabase database;

  @override
  State<ProductsPage> createState() => _ProductsPageState();
}

class _ProductsPageState extends State<ProductsPage> {
  List<Product> _products = const [];
  String _query = '';
  _ProductStatusFilter _filter = _ProductStatusFilter.active;
  Object? _loadingError;
  var _isLoading = true;
  final Set<String> _busyProductIds = {};

  @override
  void initState() {
    super.initState();
    _loadProducts();
  }

  Future<void> _loadProducts() async {
    if (mounted) {
      setState(() {
        _isLoading = true;
        _loadingError = null;
      });
    }
    try {
      final products = await widget.database.loadProducts();
      if (!mounted) return;
      setState(() {
        _products = products;
        _isLoading = false;
      });
    } catch (error) {
      if (!mounted) return;
      setState(() {
        _loadingError = error;
        _isLoading = false;
      });
    }
  }

  List<Product> get _visibleProducts {
    final query = _query.trim().toLowerCase();
    return _products.where((product) {
      final matchesStatus = switch (_filter) {
        _ProductStatusFilter.active => product.isActive,
        _ProductStatusFilter.inactive => !product.isActive,
        _ProductStatusFilter.all => true,
      };
      final matchesQuery =
          query.isEmpty ||
          product.name.toLowerCase().contains(query) ||
          product.sku.toLowerCase().contains(query) ||
          product.barcode.contains(query) ||
          product.category.toLowerCase().contains(query);
      return matchesStatus && matchesQuery;
    }).toList();
  }

  Future<void> _openEditor([Product? product]) async {
    final savedProduct = await showModalBottomSheet<Product>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) =>
          _ProductEditorSheet(database: widget.database, product: product),
    );
    if (!mounted || savedProduct == null) return;
    await _loadProducts();
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(
          product == null
              ? 'Produk berhasil ditambahkan.'
              : 'Produk berhasil diperbarui.',
        ),
      ),
    );
    if (product == null) {
      await _openLabel(savedProduct, isNewProduct: true);
    }
  }

  Future<void> _openLabel(Product product, {bool isNewProduct = false}) async {
    await showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      backgroundColor: Colors.white,
      builder: (_) => _ProductLabelSheet(
        database: widget.database,
        product: product,
        isNewProduct: isNewProduct,
      ),
    );
  }

  Future<void> _setActive(Product product, bool isActive) async {
    setState(() => _busyProductIds.add(product.id));
    try {
      await widget.database.setProductActive(
        productId: product.id,
        isActive: isActive,
      );
      await _loadProducts();
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            isActive
                ? '${product.name} diaktifkan.'
                : '${product.name} dinonaktifkan.',
          ),
        ),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Status produk belum dapat diubah.')),
      );
    } finally {
      if (mounted) setState(() => _busyProductIds.remove(product.id));
    }
  }

  Future<void> _deleteProduct(Product product) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        icon: const Icon(Icons.delete_outline, color: ZeColors.danger),
        title: const Text('Hapus produk?'),
        content: Text(
          '${product.name} akan dihapus dari katalog. Produk dengan riwayat stok atau transaksi tidak dapat dihapus.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Batal'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            style: FilledButton.styleFrom(backgroundColor: ZeColors.danger),
            child: const Text('Hapus'),
          ),
        ],
      ),
    );
    if (!mounted || confirmed != true) return;

    setState(() => _busyProductIds.add(product.id));
    try {
      await widget.database.deleteProduct(product);
      await _loadProducts();
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('${product.name} dihapus dari katalog.')),
      );
    } on ProductHasHistoryException {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            '${product.name} sudah memiliki riwayat stok atau transaksi. Nonaktifkan produk ini saja.',
          ),
        ),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Produk belum dapat dihapus.')),
      );
    } finally {
      if (mounted) setState(() => _busyProductIds.remove(product.id));
    }
  }

  @override
  Widget build(BuildContext context) {
    return _PageFrame(
      title: 'Produk',
      subtitle: 'Kelola katalog yang tampil di halaman Kasir.',
      trailing: FilledButton.icon(
        onPressed: () => _openEditor(),
        icon: const Icon(Icons.add),
        label: const Text('Tambah'),
      ),
      child: _isLoading
          ? const _ProductsLoadingState()
          : _loadingError != null
          ? _ProductsErrorState(onRetry: _loadProducts)
          : Padding(
              padding: ZeSpace.page,
              child: Column(
                children: [
                  _ProductsToolbar(
                    filter: _filter,
                    onFilterChanged: (filter) =>
                        setState(() => _filter = filter),
                    onSearchChanged: (query) => setState(() => _query = query),
                  ),
                  const SizedBox(height: 16),
                  Expanded(
                    child: _ProductList(
                      products: _visibleProducts,
                      isBusy: _busyProductIds.contains,
                      onEdit: _openEditor,
                      onLabel: _openLabel,
                      onSetActive: _setActive,
                      onDelete: _deleteProduct,
                    ),
                  ),
                ],
              ),
            ),
    );
  }
}

class _ProductLabelSheet extends StatefulWidget {
  const _ProductLabelSheet({
    required this.database,
    required this.product,
    required this.isNewProduct,
  });

  final ZePosDatabase database;
  final Product product;
  final bool isNewProduct;

  @override
  State<_ProductLabelSheet> createState() => _ProductLabelSheetState();
}

class _ProductLabelSheetState extends State<_ProductLabelSheet> {
  late Future<StoreProfile?> _profileFuture;
  late Future<PrintPurposeConfiguration> _configurationFuture;
  late final TextEditingController _copiesController;
  var _isProcessing = false;
  var _copies = 1;

  @override
  void initState() {
    super.initState();
    _profileFuture = widget.database.loadStoreProfile();
    _configurationFuture = widget.database.loadPrintConfiguration(
      PrintPurpose.label,
    );
    _copiesController = TextEditingController(text: '1');
  }

  @override
  void dispose() {
    _copiesController.dispose();
    super.dispose();
  }

  void _setCopies(int value) {
    final safeValue = value.clamp(1, 500);
    final text = '$safeValue';
    setState(() {
      _copies = safeValue;
      _copiesController.value = TextEditingValue(
        text: text,
        selection: TextSelection.collapsed(offset: text.length),
      );
    });
  }

  int? _validatedCopies() {
    final value = int.tryParse(_copiesController.text.trim());
    if (value == null || value < 1 || value > 500) return null;
    return value;
  }

  Future<void> _printLabel() async {
    final copies = _validatedCopies();
    if (copies == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Jumlah label harus antara 1 dan 500.')),
      );
      return;
    }
    _setCopies(copies);
    setState(() => _isProcessing = true);
    try {
      final results = await Future.wait([_profileFuture, _configurationFuture]);
      final printResult = await DocumentExportService.printProductLabel(
        product: widget.product,
        profile: results[0] as StoreProfile?,
        configuration: results[1] as PrintPurposeConfiguration,
        copies: copies,
      );
      if (!mounted) return;
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text(printResult.userMessage)));
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Label belum dapat dicetak. Coba lagi.')),
      );
    } finally {
      if (mounted) setState(() => _isProcessing = false);
    }
  }

  Future<void> _saveJpg() async {
    setState(() => _isProcessing = true);
    try {
      final results = await Future.wait([_profileFuture, _configurationFuture]);
      final saved = await DocumentExportService.saveProductLabelJpg(
        product: widget.product,
        profile: results[0] as StoreProfile?,
        media: (results[1] as PrintPurposeConfiguration).media,
      );
      if (!mounted || !saved) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Label JPG berhasil disimpan.')),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Label JPG belum dapat disimpan.')),
      );
    } finally {
      if (mounted) setState(() => _isProcessing = false);
    }
  }

      child: Align(
        alignment: Alignment.bottomCenter,
        heightFactor: 1,

          constraints: const BoxConstraints(maxWidth: 560),

                    fontSize: 22,

                  ),

                const SizedBox(height: 4),
                Text(
                  widget.isNewProduct
                      ? 'Barcode dibuat dari data produk baru. Cetak sekarang atau simpan untuk komputer lain.'
                      : 'Gunakan barcode ini untuk label fisik produk.',
                  style: const TextStyle(color: ZeColors.mutedAlt),
                ),
                const SizedBox(height: 20),
                FutureBuilder<PrintPurposeConfiguration>(
                  future: _configurationFuture,
                  builder: (context, snapshot) => _LabelMediaPreview(
                    product: widget.product,
                    media:
                        snapshot.data?.media ??
                        BuiltInPrinterProfiles.label40x20,
                  ),
                ),
                const SizedBox(height: ZeSpace.sm),
                FutureBuilder<PrintPurposeConfiguration>(
                  future: _configurationFuture,
                  builder: (context, snapshot) {
                    final configuration = snapshot.data;
                    return Column(
                      crossAxisAlignment: CrossAxisAlignment.stretch,
                      children: [
                        Text(
                          configuration == null
                              ? 'Memuat media label...'
                              : '${configuration.media.name} · ${configuration.binding?.printerName ?? 'Dialog Windows'}',
                          maxLines: 2,
                          overflow: TextOverflow.ellipsis,
                          style: ZeText.meta,
                        ),
                        const SizedBox(height: ZeSpace.md),
                        SizedBox(
                          width: double.infinity,
                          child: _LabelQuantityPicker(
                            controller: _copiesController,
                            value: _copies,
                            stock: widget.product.stock,
                            enabled: !_isProcessing,
                            onChanged: _setCopies,
                            onTyped: (value) {
                              final parsed = int.tryParse(value);
                              if (parsed != null &&
                                  parsed >= 1 &&
                                  parsed <= 500) {
                                setState(() => _copies = parsed);
                              }
                            },
                          ),
                        ),
                      ],
                    );
                  },
                ),
                const SizedBox(height: 18),
                Row(
                  children: [

                      ),
                    ),

                                height: 18,

                                  color: Colors.white,

            ),

class _BarcodePreview extends StatelessWidget {
  const _BarcodePreview({required this.data});

  final String data;

  @override
  Widget build(BuildContext context) {
    return Semantics(
      label: 'Barcode produk $data',
      child: CustomPaint(
        painter: _BarcodePreviewPainter(data),
        child: const SizedBox.expand(),
      ),
    );
  }
}

class _BarcodePreviewPainter extends CustomPainter {
  _BarcodePreviewPainter(this.data);

  final String data;

  @override
  void paint(Canvas canvas, Size size) {
    final bars = barcode.Barcode.code128().make(
      data,
      width: size.width,
      height: size.height,
      drawText: false,
    );
    final paint = Paint()
      ..color = ZeColors.ink
      ..isAntiAlias = false;
    for (final element in bars) {
      if (element is barcode.BarcodeBar && element.black) {
        canvas.drawRect(
          Rect.fromLTWH(
            element.left,
            element.top,
            element.width,
            element.height,
          ),
          paint,
        );
      }
    }
  }

  @override
  bool shouldRepaint(covariant _BarcodePreviewPainter oldDelegate) =>
      oldDelegate.data != data;
}

class _ProductEditorSheetState extends State<_ProductEditorSheet> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _nameController;
  late final TextEditingController _skuController;
  late final TextEditingController _barcodeController;
  late final TextEditingController _categoryController;
  late final TextEditingController _costPriceController;
  late final TextEditingController _priceController;
  late final TextEditingController _stockController;
  late final TextEditingController _minimumStockController;
  late bool _isActive;
  var _isSaving = false;
  String? _saveError;

  bool get _isEditing => widget.product != null;

  @override
  void initState() {
    super.initState();
    final product = widget.product;
    _nameController = TextEditingController(text: product?.name ?? '');
    _skuController = TextEditingController(text: product?.sku ?? '');
    _barcodeController = TextEditingController(
      text: product?.barcode.isNotEmpty == true
          ? product!.barcode
          : generateProductBarcode(),
    );
    _categoryController = TextEditingController(text: product?.category ?? '');
    _costPriceController = TextEditingController(
      text: product?.costPrice.toString() ?? '0',
    );
    _priceController = TextEditingController(
      text: product?.price.toString() ?? '',
    );
    _stockController = TextEditingController(
      text: product?.stock.toString() ?? '0',
    );
    _minimumStockController = TextEditingController(
      text: product?.minimumStock.toString() ?? '5',
    );
    _isActive = product?.isActive ?? true;
  }

  @override
  void dispose() {
    _nameController.dispose();
    _skuController.dispose();
    _barcodeController.dispose();
    _categoryController.dispose();
    _costPriceController.dispose();
    _priceController.dispose();
    _stockController.dispose();
    _minimumStockController.dispose();
    super.dispose();
  }

  void _generateBarcode() {
    setState(() => _barcodeController.text = generateProductBarcode());
  }

  String? _requiredValidator(String? value, String field) {
    if (value == null || value.trim().isEmpty) return '$field wajib diisi.';
    return null;
  }

  String? _nonNegativeNumberValidator(String? value, String field) {
    final number = int.tryParse(value?.trim() ?? '');
    if (number == null || number < 0) {
      return '$field harus angka nol atau lebih.';
    }
    return null;
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() {
      _isSaving = true;
      _saveError = null;
    });
    final oldProduct = widget.product;
    final product = Product(
      id: oldProduct?.id ?? 'product_${DateTime.now().microsecondsSinceEpoch}',
      name: _nameController.text.trim(),
      sku: _skuController.text.trim(),
      barcode: _barcodeController.text.trim(),
      category: _categoryController.text.trim(),
      costPrice: int.parse(_costPriceController.text.trim()),
      price: int.parse(_priceController.text.trim()),
      stock: oldProduct?.stock ?? int.parse(_stockController.text.trim()),
      minimumStock: int.parse(_minimumStockController.text.trim()),
      isActive: _isActive,
    );
    try {
      await widget.database.saveProduct(product);
      if (!mounted) return;
      Navigator.pop(context, product);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSaving = false;
        _saveError =
            'Produk belum disimpan. Pastikan SKU dan barcode belum digunakan.';
      });
    }
  }

String generateProductBarcode([DateTime? timestamp]) {
  final now = timestamp ?? DateTime.now();
  final serial = now.microsecondsSinceEpoch
      .remainder(1000000000)
      .toString()
      .padLeft(9, '0');
  final body = '899$serial';
  var sum = 0;
  for (var index = 0; index < body.length; index++) {
    final digit = int.parse(body[index]);
    sum += digit * (index.isEven ? 1 : 3);
  }
  final checkDigit = (10 - sum % 10) % 10;
  return '$body$checkDigit';
}

String formatRupiah(int amount) {
  final digits = amount.toString();
  final buffer = StringBuffer();
  for (var index = 0; index < digits.length; index++) {
    final positionFromEnd = digits.length - index;
    buffer.write(digits[index]);
    if (positionFromEnd > 1 && positionFromEnd % 3 == 1) {
      buffer.write('.');
    }
  }
  return 'Rp$buffer';
}

String formatDateTime(DateTime value) {
  const months = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'Mei',
    'Jun',
    'Jul',
    'Agu',
    'Sep',
    'Okt',
    'Nov',
    'Des',
  ];
  final minute = value.minute.toString().padLeft(2, '0');
  return '${value.day} ${months[value.month - 1]} ${value.year}, ${value.hour}:$minute';
}

String formatReportPeriod(DateTime start, DateTime end) {
  if (start.year == end.year &&
      start.month == end.month &&
      start.day == end.day) {
    return 'Laporan ${formatDateTime(start).split(',').first}';
  }
  const months = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'Mei',
    'Jun',
    'Jul',
    'Agu',
    'Sep',
    'Okt',
    'Nov',
    'Des',
  ];
  return '${start.day} ${months[start.month - 1]} ${start.year} – ${end.day} ${months[end.month - 1]} ${end.year}';
}

String formatFileSize(int sizeBytes) {
  if (sizeBytes < 1024) return '$sizeBytes B';
  final kilobytes = sizeBytes / 1024;
  if (kilobytes < 1024) return '${kilobytes.toStringAsFixed(1)} KB';
  return '${(kilobytes / 1024).toStringAsFixed(1)} MB';
}
